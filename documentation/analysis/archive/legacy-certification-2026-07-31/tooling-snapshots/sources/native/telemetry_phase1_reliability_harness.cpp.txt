	detail::ProducerSessionProgress progress,
	std::uint64_t slot_session_id,
	std::uint64_t peer_session_id) noexcept {
	return (progress == detail::ProducerSessionProgress::ReadyForState ||
			   progress == detail::ProducerSessionProgress::Stale) &&
		slot_session_id != 0U &&
		peer_session_id == slot_session_id;
}

	bool resync_sent,
	bool recovered_snapshot,
	std::uint32_t initial_snapshot_id,
	std::uint32_t producer_snapshot_id) noexcept {
	return resync_sent &&
		recovered_snapshot &&
		initial_snapshot_id != 0U &&
		producer_snapshot_id != 0U &&
		producer_snapshot_id != initial_snapshot_id;
}

	return !resync_sent && transport_drained;

	bool t0_established, bool post_t0_resync_sent,
	bool final_resync_armed) noexcept {
	return (!t0_established && !final_resync_armed) ||
		(t0_established && !post_t0_resync_sent);

	std::uint32_t producer_snapshot_id = 0U;

bool phase2_terminal_resync_stable_for_two_turns(
	bool candidate,
	const Phase2TerminalResyncKey& current,
	Phase2TerminalResyncKey& previous,
	std::uint32_t& stable_turns) noexcept {
	const bool same =
		current.session_id == previous.session_id &&
		current.message_id == previous.message_id &&
		current.initial_snapshot_id ==
			previous.initial_snapshot_id &&
		current.producer_snapshot_id ==
			previous.producer_snapshot_id &&
		current.retries == previous.retries;
	if (!candidate) {
		previous = {};
		stable_turns = 0U;
		return false;
	}
	if (!same) {
		previous = current;
		stable_turns = 1U;
		return false;
	}
	if (stable_turns < 2U)
		++stable_turns;
	return stable_turns >= 2U;
}

enum class Phase2FinalResyncRearm {
	Unrelated,
	Released,
	Exhausted
};

enum class Phase2RecoveryExitReason : std::uint8_t {
	None = 0U,
	Converged = 1U,
	DeadlineExpired = 2U,
	RearmExhausted = 3U
};

std::uint64_t phase2_saturating_deadline(
	std::uint64_t start_us, std::uint64_t duration_us) noexcept {
	return start_us >
			std::numeric_limits<std::uint64_t>::max() - duration_us
		? std::numeric_limits<std::uint64_t>::max()
		: start_us + duration_us;
}

std::uint64_t phase2_final_resync_attempt_deadline(
	std::uint64_t arm_us) noexcept {
	return phase2_saturating_deadline(
		arm_us, Phase2FinalResyncRetryEnvelopeUs);
}

std::uint64_t phase2_post_resync_heartbeat_deadline(
	std::uint64_t snapshot_published_us,
	std::uint16_t heartbeat_interval_ms) noexcept {
	const auto heartbeat_interval_us =
		static_cast<std::uint64_t>(heartbeat_interval_ms) * 1000U;
	return phase2_saturating_deadline(
		snapshot_published_us,
		heartbeat_interval_us +
			Phase2MaximumDeliveryDelayUs +
			4U * Phase2RecoveryTickUs);
}

std::uint64_t phase2_post_t0_deadline(std::uint64_t t0_us) noexcept {
	return phase2_saturating_deadline(
		t0_us, Phase2PostT0ConvergenceUs);
}

bool phase2_recovery_deadline_expired(
	std::uint64_t now_us, std::uint64_t deadline_us) noexcept {
	return now_us > deadline_us;
}

bool phase2_advance_recovery_clock(
	std::uint64_t now_us,
	std::uint64_t deadline_us,
	std::uint64_t& next_us) noexcept {
	next_us = now_us;
	if (now_us >= deadline_us)
		return false;
	next_us = phase2_saturating_deadline(
		now_us, Phase2RecoveryTickUs);
	if (next_us > deadline_us)
		next_us = deadline_us;
	return next_us > now_us;
}

Phase2FinalResyncRearm
phase2_release_final_resync_after_terminal_abandon(
	std::uint64_t abandoned_session_id,
	std::uint64_t& final_resync_session_id,
	std::uint64_t& final_resync_ack_tick,
	std::uint64_t& final_resync_ack_clock_response_us,
	bool& rearm_used) noexcept {
	if (final_resync_session_id == 0U ||
		final_resync_session_id != abandoned_session_id)
		return Phase2FinalResyncRearm::Unrelated;
	if (rearm_used)
		return Phase2FinalResyncRearm::Exhausted;
	rearm_used = true;
	final_resync_session_id = 0U;
	final_resync_ack_tick =
		std::numeric_limits<std::uint64_t>::max();
	final_resync_ack_clock_response_us = 0U;
	return Phase2FinalResyncRearm::Released;
}

bool phase2_blackout_preconditioning_due(
	const detail::SessionControllerSlot& slot,
	std::uint64_t now_us,
	std::uint64_t blackout_start_us) noexcept {
	if (slot.heartbeat.last_valid_network_activity_us == 0U ||
		slot.heartbeat.disconnect_timeout_us == 0U ||
		slot.heartbeat.last_valid_network_activity_us >
			std::numeric_limits<std::uint64_t>::max() -
				slot.heartbeat.disconnect_timeout_us)
		return false;
	const auto disconnect_deadline_us =
		slot.heartbeat.last_valid_network_activity_us +
		slot.heartbeat.disconnect_timeout_us;
	constexpr auto scheduling_horizon_us =
		Phase2MaximumDeliveryDelayUs +
		Phase2BlackoutProbeSpacingUs;
	return disconnect_deadline_us <= blackout_start_us &&
		now_us < disconnect_deadline_us &&
		disconnect_deadline_us - now_us <= scheduling_horizon_us;
}

bool phase2_blackout_lead_in_emission_due(
	bool preconditioning_due,
	bool in_lead_in,
	bool lead_in_scheduled,
	bool response_coverage_complete) noexcept {
	return !lead_in_scheduled &&
		(preconditioning_due ||
		 (in_lead_in && !response_coverage_complete));
}

std::uint64_t phase2_blackout_start_us(std::size_t window) noexcept {
	if (!phase2_evidence_configuration.proportional_blackouts ||
		phase2_evidence_configuration.measured_duration_us == 0U)
		return window == 0U ? 120000000ULL : 360000000ULL;
	return phase2_evidence_configuration.measured_duration_us *
		(window == 0U ? 40ULL : 75ULL) / 100ULL;
}

void phase2_record_final_resync_ack_decision(
	std::uint32_t target_message_id, bool dropped) noexcept {
	const auto arm = std::find(
		phase2_evidence_result.final_resync_message_id_at_arm.begin(),
		phase2_evidence_result.final_resync_message_id_at_arm.end(),
		target_message_id);
	if (target_message_id == 0U ||
		arm ==
			phase2_evidence_result.final_resync_message_id_at_arm.end())
		return;
	const auto index = static_cast<std::size_t>(
		std::distance(
			phase2_evidence_result
				.final_resync_message_id_at_arm.begin(),
			arm));
	if (phase2_evidence_result
			.validated_resync_ack_target_message_ids[index] == 0U) {
		phase2_evidence_result
			.validated_resync_ack_target_message_ids[index] =
			target_message_id;
		++phase2_evidence_result
			  .validated_resync_ack_decision_count;
	}
	phase2_evidence_result
		.validated_resync_ack_any_delivered[index] =
		phase2_evidence_result
			.validated_resync_ack_any_delivered[index] ||
		!dropped;
	phase2_evidence_result
		.validated_resync_ack_any_dropped[index] =
		phase2_evidence_result
			.validated_resync_ack_any_dropped[index] ||
		dropped;
}

struct Phase2EvidenceRandom {
	std::uint64_t state = 0U;
	std::uint64_t next() noexcept {
		state += 0x9e3779b97f4a7c15ULL;
		auto value = state;
		value = (value ^ (value >> 30U)) * 0xbf58476d1ce4e5b9ULL;
		value = (value ^ (value >> 27U)) * 0x94d049bb133111ebULL;
		return value ^ (value >> 31U);
	}
	std::uint32_t bounded(std::uint32_t bound) noexcept {
		return bound == 0U ? 0U : static_cast<std::uint32_t>(next() % bound);
	}
};

std::uint64_t phase2_stable_hash(const char* text) noexcept {
	std::uint64_t value = 1469598103934665603ULL;
	for (const auto* cursor = text; cursor != nullptr && *cursor; ++cursor) {
		value ^= static_cast<unsigned char>(*cursor);
		value *= 1099511628211ULL;
	}
	return value;
}

std::string phase2_state_sha256(const protocol::StateImage& image) {
	std::vector<std::uint8_t> canonical;
	auto append_u16 = [&](std::uint16_t value) {
		canonical.push_back(static_cast<std::uint8_t>(value & 0xffU));
		canonical.push_back(static_cast<std::uint8_t>((value >> 8U) & 0xffU));
	};
	auto append_u32 = [&](std::uint32_t value) {
		for (std::uint32_t shift = 0U; shift != 32U; shift += 8U)
			canonical.push_back(static_cast<std::uint8_t>((value >> shift) & 0xffU));
	};
	for (const auto& atom : image.records()) {
		append_u16(atom.key.record_type);
		append_u32(static_cast<std::uint32_t>(atom.key.identity.size()));
		canonical.insert(canonical.end(), atom.key.identity.begin(), atom.key.identity.end());
		canonical.push_back(atom.record_version);
		canonical.push_back(static_cast<std::uint8_t>(atom.lifecycle));
		canonical.push_back(atom.has_cascade_owner ? 1U : 0U);
		append_u16(atom.cascade_owner.record_type);
		append_u32(static_cast<std::uint32_t>(atom.cascade_owner.identity.size()));
		canonical.insert(canonical.end(), atom.cascade_owner.identity.begin(),
			atom.cascade_owner.identity.end());
		append_u32(static_cast<std::uint32_t>(atom.value.size()));
		canonical.insert(canonical.end(), atom.value.begin(), atom.value.end());
	}
	protocol::Sha256Digest digest{};
	if (!protocol::sha256({canonical.data(), canonical.size()}, digest))
		return {};
	std::ostringstream output;
	for (const auto byte : digest)
		output << std::hex << std::setfill('0') << std::setw(2)
			   << static_cast<unsigned>(byte);
	return output.str();
}

	std::uint64_t send_attempts = 0U, loss = 0U, duplicate_delivered = 0U, reorder_delivered = 0U, jitter = 0U, fragment_probe = 0U;
	std::uint32_t loss_percent = 1U; bool burst = false, impairments_active = false;
	std::uint64_t now_us = 0U, loss_accumulator = 0U, burst_remaining = 0U;
	std::uint64_t delta_loss = 0U, snapshot_fragment_loss = 0U, would_block_injected = 0U, outage_loss = 0U, jitter_delayed = 0U;

	// reference impairment.  It is deliberately distinct from the continuous
	// configured loss, whose end is not a meaningful recovery boundary.

	// Complete while withholding this datagram models a real downstream UDP
	// loss: the production reliable window must retransmit it before it can be
	// ACKed.  Handshake control is never synthetically lost.
	std::uint32_t recovery_losses_remaining = 0U;

		std::uint64_t order = 0U;

			protocol::MessageType::Invalid;

	struct DirectionState {

		Phase2EvidenceRandom duplication{};

		std::uint32_t burst_start = 0U;

	void configure_phase2() {
		phase2_measurement_started =
			phase2_evidence_configuration.warmup_us != 0U;
		for (std::size_t direction = 0U; direction < phase2_directions.size(); ++direction) {
			const auto label = direction == 0U ? "producer-to-client" : "client-to-producer";
			const auto prefix = std::string(phase2_evidence_configuration.profile) + "|" +
				std::to_string(phase2_evidence_configuration.loss_rate) + "|" +
				phase2_evidence_configuration.mode + "|" + label;
			auto& state = phase2_directions[direction];
			state.loss.state = phase2_evidence_configuration.seed ^ phase2_stable_hash((prefix + "|loss").c_str());
			state.duplication.state = phase2_evidence_configuration.seed ^ phase2_stable_hash((prefix + "|duplication").c_str());
			state.jitter.state = phase2_evidence_configuration.seed ^ phase2_stable_hash((prefix + "|jitter").c_str());
			state.reorder.state = phase2_evidence_configuration.seed ^ phase2_stable_hash((prefix + "|reorder").c_str());
		}
		if (phase2_evidence_configuration.decision_log != nullptr)
			phase2_decisions.open(phase2_evidence_configuration.decision_log, std::ios::out | std::ios::trunc);
	}

	void phase2_flush(std::vector<ScheduledPacket>& scheduled, std::vector<Packet>& output) {
		std::stable_sort(scheduled.begin(), scheduled.end(), [](const auto& left, const auto& right) {
			return left.due_us < right.due_us ||
				(left.due_us == right.due_us && left.order < right.order);
		});
		auto cursor = scheduled.begin();
		while (cursor != scheduled.end() && cursor->due_us <= now_us) {
			output.push_back(std::move(cursor->packet));
			++cursor;
		}
		scheduled.erase(scheduled.begin(), cursor);
	}

	bool phase2_schedule(Packet packet, std::size_t direction, protocol::MessageType message_type) {
		const auto record_validated_resync_ack =
			[&](bool dropped) {
				if (direction != 0U ||
					message_type != protocol::MessageType::Ack)
					return;
				protocol::DatagramView ack_datagram{};
				protocol::AckPayload ack{};
				if (protocol::decode_and_validate_datagram(
						{packet.data.data(), packet.data.size()},
						{protocol::VersionMinorV1_1,
						 protocol::VersionMinorV1_1},
						ack_datagram) ==
						protocol::ValidationError::None &&
					protocol::decode_ack_payload(
						ack_datagram.payload, ack) ==
						protocol::ValidationError::None &&
					ack.target_message_type ==
						protocol::MessageType::ResyncRequest &&
					(ack.ack_flags &
					 static_cast<std::uint8_t>(
						 protocol::AckFlag::Validated)) != 0U) {
					phase2_record_final_resync_ack_decision(
						ack.target_message_id, dropped);
				}
			};
		auto& state = phase2_directions[direction];
		if (!phase2_measurement_started) {
			record_validated_resync_ack(false);
			(direction == 0U ? tx : rx).push_back(std::move(packet));
			return true;
		}
		const auto measured_start_us = phase2_campaign_start_us + phase2_evidence_configuration.warmup_us;
		if (now_us < measured_start_us) {
			record_validated_resync_ack(false);
			(direction == 0U ? tx : rx).push_back(std::move(packet));
			return true;
		}
		const auto measured_us = now_us - measured_start_us;
		const auto ordinal = state.ordinal++;
		const auto block = ordinal / 100U;
		if (block != state.block) {
			state.block = block;
			state.burst_start = state.loss.bounded(100U);
		}
		const auto position = static_cast<std::uint32_t>(ordinal % 100U);
		const auto relative = static_cast<std::uint32_t>((position + 100U - state.burst_start) % 100U);
		const bool algorithm_drop = std::strcmp(phase2_evidence_configuration.mode, "burst") == 0
			? relative < phase2_evidence_configuration.loss_rate
			: state.loss.bounded(1000000U) < phase2_evidence_configuration.loss_rate * 10000U;
		std::size_t blackout_window = 2U;
		for (std::size_t window = 0U; window < 2U; ++window) {
			const auto start_us = phase2_blackout_start_us(window);
			if (measured_us >= start_us &&
				measured_us < start_us + Phase2BlackoutDurationUs) {
				blackout_window = window;
				break;
			}
		}
		const bool blackout = blackout_window < 2U;
		if (blackout) {
			++phase2_evidence_result
				  .blackout_directional_decisions[blackout_window][direction];
		}
		if (message_type == protocol::MessageType::Heartbeat) {
			protocol::DatagramView datagram{};
			protocol::HeartbeatPayload heartbeat{};
			if (protocol::decode_and_validate_datagram(
					{packet.data.data(), packet.data.size()},
					{protocol::VersionMinorV1_1,
					 protocol::VersionMinorV1_1},
					datagram) == protocol::ValidationError::None &&
				protocol::decode_heartbeat_payload(
					datagram.payload, heartbeat) ==
					protocol::ValidationError::None) {
				auto probe = std::find_if(
					phase2_evidence_probes.begin(),
					phase2_evidence_probes.end(),
					[&](const auto& candidate) {
						return candidate.session_id ==
								datagram.header.session_id &&
							candidate.probe_id ==
								heartbeat.probe_id &&
							candidate.origin_t0_us ==
								heartbeat.origin_t0_us;
					});
				if (probe != phase2_evidence_probes.end() &&
					heartbeat.kind ==
						protocol::HeartbeatKind::Request &&
					direction == 1U) {
					probe->request_decision_measured_us =
						measured_us;
					if (probe->lead_in &&
						probe->window <
							phase2_evidence_result
								.blackout_lead_in_requests
								.size() &&
						!phase2_evidence_result
							 .blackout_correlated_responses[
								 probe->window]
							 .present) {
						probe->request_decision_observed =
							true;
						phase2_evidence_result
							.blackout_lead_in_requests[
								probe->window] = {
								true,
								probe->session_id,
								probe->probe_id,
								probe->origin_t0_us,
								measured_us};
					}
				}
				if (probe != phase2_evidence_probes.end() &&
					blackout && !probe->lead_in &&
					heartbeat.kind ==
						protocol::HeartbeatKind::Request &&
					direction == 1U &&
					!probe->request_decision_observed) {
					probe->request_decision_observed = true;
					phase2_evidence_result
						.blackout_in_window_requests[
							blackout_window] = {
							true,
							datagram.header.session_id,
							heartbeat.probe_id,
							heartbeat.origin_t0_us,
							measured_us};
					++phase2_evidence_result
						  .blackout_probe_request_decisions[
							  blackout_window];
				}
				if (probe != phase2_evidence_probes.end() &&
					blackout && probe->lead_in &&
					heartbeat.kind ==
						protocol::HeartbeatKind::Response &&
					direction == 0U &&
					!probe->response_decision_observed &&
					probe->window == blackout_window &&
					!phase2_evidence_result
						 .blackout_correlated_responses[
							 blackout_window]
						 .present) {
					probe->response_decision_observed = true;
					phase2_evidence_result
						.blackout_lead_in_requests[
							blackout_window] = {
							true,
							probe->session_id,
							probe->probe_id,
							probe->origin_t0_us,
							probe
								->request_decision_measured_us};
					phase2_evidence_result
						.blackout_correlated_responses[
							blackout_window] = {
							true,
							datagram.header.session_id,
							heartbeat.probe_id,
							heartbeat.origin_t0_us,
							measured_us,
							heartbeat.receive_t1_us,
							heartbeat.transmit_t2_us,
							datagram.header.sent_time_us};
					++phase2_evidence_result
						  .blackout_probe_response_decisions[
							  blackout_window];
				}
			}
		}
		const bool drop = algorithm_drop || blackout;
		record_validated_resync_ack(drop);
		const bool duplicate = state.duplication.bounded(100U) < 2U;
		const auto jitter_ms = state.jitter.bounded(101U);
		const auto reorder_slot = state.reorder.bounded(9U);
		const auto due_us = now_us + static_cast<std::uint64_t>(jitter_ms) * 1000ULL +
			static_cast<std::uint64_t>(reorder_slot) * 33334ULL;
		++phase2_evidence_result.decision_records;
		++phase2_evidence_result.directional_records[direction];
		phase2_evidence_result.directional_drops[direction] += drop ? 1U : 0U;
		if (phase2_decisions) {
			phase2_decisions << "{\"schema\":\"fs2open.telemetry.phase2.loss-decision.v1\","
				<< "\"profile\":\"" << phase2_evidence_configuration.profile << "\",\"rate\":"
				<< phase2_evidence_configuration.loss_rate << ",\"mode\":\""
				<< phase2_evidence_configuration.mode << "\",\"direction\":\""
				<< (direction == 0U ? "producer-to-client" : "client-to-producer")
				<< "\",\"streamKey\":\"" << phase2_evidence_configuration.profile << "|"
				<< phase2_evidence_configuration.loss_rate << "|" << phase2_evidence_configuration.mode << "|"
				<< (direction == 0U ? "producer-to-client" : "client-to-producer")
				<< "\",\"ordinal\":" << ordinal << ",\"measuredUs\":" << measured_us
				<< ",\"block\":" << block << ",\"position\":" << position
				<< ",\"burstStart\":" << state.burst_start
				<< ",\"algorithmDrop\":" << (algorithm_drop ? "true" : "false")
				<< ",\"blackout\":" << (blackout ? "true" : "false")
				<< ",\"drop\":" << (drop ? "true" : "false")
				<< ",\"duplicate\":" << (duplicate ? "true" : "false")
				<< ",\"jitterMs\":" << jitter_ms << ",\"reorderSlot\":" << reorder_slot
				<< ",\"messageType\":" << static_cast<unsigned>(message_type) << "}\n";
		}
		if (drop) return true;
		auto& scheduled = direction == 0U ? phase2_downstream : phase2_upstream;
		scheduled.push_back({
			due_us, phase2_schedule_order++, packet,
			message_type});
		if (duplicate)
			scheduled.push_back({
				due_us + 1U, phase2_schedule_order++,
				std::move(packet), message_type});
		return true;
	}

	void phase2_queue_peer(Packet packet) {
		protocol::DatagramView datagram{};
		const auto decoded = protocol::decode_and_validate_datagram(bytes(packet.data),
			{protocol::VersionMinorV1_1, protocol::VersionMinorV1_1}, datagram);
		if (decoded != protocol::ValidationError::None) return;
		(void)phase2_schedule(std::move(packet), 1U, datagram.header.message_type);
	}
	void phase2_flush_downstream() {
		phase2_flush(phase2_downstream, tx);
	}
#endif
	bool outage_active() const noexcept { const auto second=(now_us/1000000U)%60U; return second>=30U && second<32U; }
	void observe_bounded_impairment_window() noexcept {
		if (!impairments_active || bounded_impairment_start_us != 0U || !outage_active()) return;
		const auto minute_start_us = (now_us / 60000000U) * 60000000U;
		bounded_impairment_start_us = minute_start_us + 30000000U;
		bounded_impairment_end_us = minute_start_us + 32000000U;
	}
	detail::SocketOpenResult open_socket(const detail::SocketOpenRequest& request) noexcept override {
		return {detail::SocketOpenStatus::Complete,1U,protocol::EndpointKey::from_ipv4({127U,0U,0U,1U},request.port)};
	}
	detail::SocketReceiveResult try_receive(detail::SocketHandle, protocol::MutableByteView out) noexcept override {
#if defined(FSO_TELEMETRY_PHASE2_EVIDENCE)
		phase2_flush(phase2_upstream, rx);
#endif
		++jitter;
		if (impairments_active && rx_index < rx.size() && (jitter % 17U)==0U) { ++jitter_delayed; return {detail::IoStatus::WouldBlock,{},0U,false}; }
		if (rx_index >= rx.size()) { ++fragment_probe; return {detail::IoStatus::WouldBlock,{},0U,false}; }
		const auto& packet = rx[rx_index++];
		if (packet.data.size() > out.size) return {detail::IoStatus::Error,{},0U,true};
		std::copy(packet.data.begin(),packet.data.end(),out.data);
		return {detail::IoStatus::Complete,packet.endpoint,packet.data.size(),false};
	}
	detail::SocketSendResult try_send(detail::SocketHandle, const protocol::EndpointKey& endpoint, protocol::ByteView payload) noexcept override {
		++send_attempts;
		protocol::DatagramView datagram{};
		const auto decoded = protocol::decode_and_validate_datagram(payload,{protocol::VersionMinorV1_1,protocol::VersionMinorV1_1},datagram)==protocol::ValidationError::None;
#if defined(FSO_TELEMETRY_PHASE2_EVIDENCE)
		phase2_flush(phase2_downstream, tx);
		if (decoded) {
			Packet packet{{payload.data, payload.data + payload.size}, endpoint};
			(void)phase2_schedule(std::move(packet), 0U, datagram.header.message_type);
			return {detail::IoStatus::Complete, payload.size};
		}
#endif
		if (recovery_losses_remaining != 0U && decoded && datagram.header.message_type==protocol::MessageType::FullSnapshot) {
			--recovery_losses_remaining; ++loss; ++snapshot_fragment_loss;
			return {detail::IoStatus::Complete,payload.size};
		}
		if (impairments_active && decoded && (datagram.header.message_type==protocol::MessageType::Delta || datagram.header.message_type==protocol::MessageType::FullSnapshot)) {
			// A two-second deterministic cut each simulated minute is real UDP
			// downstream loss, never a synthetic report-only counter.
			if (outage_active()) {
				++loss; ++outage_loss; return {detail::IoStatus::Complete,payload.size};
			}
			if (datagram.header.message_type==protocol::MessageType::Delta && (send_attempts % 47U)==0U) {
				++would_block_injected; return {detail::IoStatus::WouldBlock,0U};
			}
			bool drop = false;
			if (burst) {
				if (burst_remaining != 0U) { --burst_remaining; drop = true; }
				else { loss_accumulator += loss_percent; if (loss_accumulator >= 300U) { loss_accumulator -= 300U; burst_remaining=2U; drop=true; } }
			} else { loss_accumulator += loss_percent; if (loss_accumulator >= 100U) { loss_accumulator -= 100U; drop=true; } }
			if (drop) { ++loss; if(datagram.header.message_type==protocol::MessageType::Delta)++delta_loss; else ++snapshot_fragment_loss; return {detail::IoStatus::Complete,payload.size}; }
		}
		tx.push_back({{payload.data,payload.data + payload.size},endpoint});
		return {detail::IoStatus::Complete,payload.size};
	}
	// Fully encoded peer frames are actually queued through the production
	// receive path. They are not merely observed counters.

		phase2_queue_peer(std::move(packet));
#else
		rx.push_back(packet);
		rx.push_back(std::move(packet));
		++duplicate_delivered;
#endif
	}
	void enqueue_reordered(Packet current, Packet delayed) {
#if defined(FSO_TELEMETRY_PHASE2_EVIDENCE)
		phase2_queue_peer(std::move(current));
		phase2_queue_peer(std::move(delayed));
#else
		rx.push_back(std::move(current));
		rx.push_back(std::move(delayed));
		++reorder_delivered;
#endif
	}
	void enqueue_peer(Packet packet) {
#if defined(FSO_TELEMETRY_PHASE2_EVIDENCE)
		phase2_queue_peer(std::move(packet));
#else

#endif
	}

	bool phase2_has_lead_in_heartbeat_due_in(
		std::uint64_t start_us, std::uint64_t end_us,
		std::size_t window,
		std::uint64_t session_id) const noexcept {
		return std::any_of(
			phase2_upstream.begin(), phase2_upstream.end(),
			[start_us, end_us, window, session_id](
				const auto& scheduled) {
				if (scheduled.due_us < start_us ||
					scheduled.due_us >= end_us ||
					scheduled.message_type !=
						protocol::MessageType::Heartbeat)
					return false;
				protocol::DatagramView datagram{};
				protocol::HeartbeatPayload heartbeat{};
				if (protocol::decode_and_validate_datagram(
						{scheduled.packet.data.data(),
						 scheduled.packet.data.size()},
						{protocol::VersionMinorV1_1,
						 protocol::VersionMinorV1_1},
						datagram) !=
						protocol::ValidationError::None ||
					protocol::decode_heartbeat_payload(
						datagram.payload, heartbeat) !=
						protocol::ValidationError::None ||
					heartbeat.kind !=
						protocol::HeartbeatKind::Request ||
					datagram.header.session_id != session_id)
					return false;
				return std::any_of(
					phase2_evidence_probes.begin(),
					phase2_evidence_probes.end(),
					[&](const auto& probe) {
						return probe.lead_in &&
							probe.window == window &&
							probe.session_id ==
								session_id &&
							probe.probe_id ==
								heartbeat.probe_id &&
							probe.origin_t0_us ==
								heartbeat.origin_t0_us;
					});
			});
	}
	bool phase2_has_pending_resync_exchange(
		std::uint64_t session_id,
		std::uint32_t resync_message_id,
		std::size_t peer_tx_seen) const noexcept {
		const auto related = [&](const Packet& packet) {
			protocol::DatagramView datagram{};
			if (protocol::decode_and_validate_datagram(
					{packet.data.data(), packet.data.size()},
					{protocol::VersionMinorV1_1,
					 protocol::VersionMinorV1_1},
					datagram) != protocol::ValidationError::None ||
				datagram.header.session_id != session_id)
				return false;
			if (datagram.header.message_type ==
					protocol::MessageType::ResyncRequest)
				return datagram.header.message_id ==
					resync_message_id;
			if (datagram.header.message_type ==
					protocol::MessageType::FullSnapshot ||
				datagram.header.message_type ==
					protocol::MessageType::Manifest)
				return true;
			if (datagram.header.message_type !=
					protocol::MessageType::Ack)
				return false;
			protocol::AckPayload ack{};
			if (protocol::decode_ack_payload(
					datagram.payload, ack) !=
					protocol::ValidationError::None)
				return true;
			return (ack.target_message_type ==
						protocol::MessageType::ResyncRequest &&
					ack.target_message_id ==
						resync_message_id) ||
				ack.target_message_type ==
					protocol::MessageType::FullSnapshot ||
				ack.target_message_type ==
					protocol::MessageType::Manifest;
		};
		const auto scheduled_related =
			[&](const auto& scheduled) {
				return related(scheduled.packet);
			};
		if (std::any_of(
				phase2_upstream.begin(), phase2_upstream.end(),
				scheduled_related) ||
			std::any_of(
				phase2_downstream.begin(), phase2_downstream.end(),
				scheduled_related))
			return true;
		for (std::size_t index = rx_index; index < rx.size(); ++index)
			if (related(rx[index]))
				return true;
		for (std::size_t index = std::min(peer_tx_seen, tx.size());
			 index < tx.size(); ++index)
			if (related(tx[index]))
				return true;
		return false;
	}
#endif
};

bool valid_profile(const char* profile, Backend& backend) noexcept {
	if (std::strncmp(profile,"independent-loss-",17) == 0) backend.burst=false;
	else if (std::strncmp(profile,"burst-loss-",11) == 0) backend.burst=true;
	else return false;
	const auto value = std::strtoul(profile + (backend.burst ? 11 : 17),nullptr,10);
	backend.loss_percent=static_cast<std::uint32_t>(value);
	return value == 1U || value == 5U || value == 20U;
}

bool make_ack(const Backend::Packet& target, std::uint32_t sequence, Backend::Packet& result) noexcept {

	bool accept_manifest(const Backend::Packet& packet,
		const protocol::DatagramView& datagram,
		std::uint64_t now_us,
		Backend& backend) {
		protocol::ManifestPartPayload part{};
		if (protocol::decode_manifest_part_payload(datagram.payload, part) != protocol::ValidationError::None)
			return false;
		if (manifest_transaction.session_id == 0U) {
			manifest_transaction.session_id = datagram.header.session_id;
			manifest_transaction.semantic_id = part.manifest_id;
			manifest_transaction.part_count = part.part_count;
			manifest_transaction.transaction_size =
				part.transaction_size;
			manifest_transaction.transaction_sha256 =
				part.transaction_sha256;
			manifest_transaction.seen.assign(part.part_count, false);
			manifest_transaction.records.resize(part.part_count);
			manifest_transaction.record_counts.assign(
				part.part_count, 0U);
		}
		if (manifest_transaction.session_id != datagram.header.session_id ||
			manifest_transaction.semantic_id != part.manifest_id ||
			manifest_transaction.part_count != part.part_count ||
			manifest_transaction.transaction_size !=
				part.transaction_size ||
			manifest_transaction.transaction_sha256 !=
				part.transaction_sha256 ||
			part.part_index >= manifest_transaction.part_count)
			return false;
		manifest_transaction.seen[part.part_index] = true;
		manifest_transaction.records[part.part_index].assign(part.records.data,
			part.records.data + part.records.size);
		manifest_transaction.record_counts[part.part_index] =
			part.record_count;
		manifest_transaction.record_count = static_cast<std::uint16_t>(
			manifest_transaction.record_count + part.record_count);
		if (std::find(manifest_transaction.seen.begin(), manifest_transaction.seen.end(), false) ==
			manifest_transaction.seen.end()) {
			class_ids.clear();
			weapon_class_ids.clear();
			weapon_class_flags.clear();
			for (std::size_t region_index = 0U;
				region_index <
					manifest_transaction.records.size();
				++region_index) {
				const auto& region =
					manifest_transaction.records[region_index];
				protocol::RecordEnvelopeIterator iterator(
					{region.data(), region.size()},
					manifest_transaction.record_counts[
						region_index],
					protocol::RecordFlagPolicy::RequireNone);
				for (;;) {
					protocol::RecordEnvelopeView record{};
					bool has_value = false;
					if (iterator.next(record, has_value) !=
						protocol::ValidationError::None)
						return false;
					if (!has_value) break;
					protocol::BusinessRecordMetadata metadata{};
					if (protocol::validate_business_record(
							record,
							protocol::BusinessRecordContainer::Manifest,
							datagram.header.version_minor,
							metadata) !=
						protocol::ValidationError::None)
						return false;
					protocol::PacketReader reader(record.payload);
					std::uint32_t generation = 0U;
					std::uint32_t public_id = 0U;
					if (!reader.read_u32(generation) ||
						!reader.read_u32(public_id))
						return false;
					if (metadata.type ==
						protocol::RecordType::ClassManifest)
						class_ids.push_back(public_id);
					else if (metadata.type ==
						protocol::RecordType::WeaponManifest) {
						weapon_class_ids.push_back(public_id);
						weapon_class_flags.push_back(0U);
					}
				}
			}
			std::sort(class_ids.begin(), class_ids.end());
			class_catalog.clear();
			for (const auto class_id : class_ids)
				class_catalog.push_back(
					{class_id, nullptr, 0U});
			const auto installed = client.install_manifest(part.manifest_id);
			if (installed != protocol::ManifestInstallResult::Installed &&
				installed != protocol::ManifestInstallResult::AlreadyInstalled)
				return false;
			Backend::Packet ack{};
			if (!make_ack(packet, sequence++, ack)) return false;
			backend.enqueue_peer(std::move(ack));
			manifest_applied_us = now_us;
			phase2_evidence_result.manifest_applied_us = now_us;
			phase2_evidence_result.manifest_installed = true;
			phase2_evidence_result.latest_manifest_id = part.manifest_id;
			++phase2_evidence_result.manifest_apply_count;
			manifest_transaction.clear();
		}
		return true;
	}

	SnapshotAcceptResult accept_snapshot_part(const protocol::FullSnapshotPartPayload& part,
		const protocol::DatagramView& datagram,
		std::uint64_t now_us) {
		if (snapshot_transaction.session_id == 0U) {
			snapshot_transaction.session_id = datagram.header.session_id;
			snapshot_transaction.semantic_id = part.snapshot_id;
			snapshot_transaction.required_manifest_id = part.required_manifest_id;
			snapshot_transaction.part_count = part.part_count;
			snapshot_transaction.transaction_size =
				part.transaction_size;
			snapshot_transaction.transaction_sha256 =
				part.transaction_sha256;
			snapshot_transaction.seen.assign(part.part_count, false);
			snapshot_transaction.records.resize(part.part_count);
			snapshot_transaction.record_counts.assign(
				part.part_count, 0U);
		}
		if (snapshot_transaction.session_id != datagram.header.session_id ||
			snapshot_transaction.semantic_id != part.snapshot_id ||
			snapshot_transaction.required_manifest_id != part.required_manifest_id ||
			snapshot_transaction.part_count != part.part_count ||
			snapshot_transaction.transaction_size !=
				part.transaction_size ||
			snapshot_transaction.transaction_sha256 !=
				part.transaction_sha256 ||
			part.part_index >= snapshot_transaction.part_count) {
			phase2_evidence_result.no_partial_publication = false;
			return SnapshotAcceptResult::Rejected;
		}
		if (!snapshot_transaction.seen[part.part_index]) {
			snapshot_transaction.record_count = static_cast<std::uint16_t>(
				snapshot_transaction.record_count + part.record_count);
		}
		snapshot_transaction.seen[part.part_index] = true;
		snapshot_transaction.records[part.part_index].assign(part.records.data,
			part.records.data + part.records.size);
		snapshot_transaction.record_counts[part.part_index] =
			part.record_count;
		if (std::find(snapshot_transaction.seen.begin(), snapshot_transaction.seen.end(), false) !=
			snapshot_transaction.seen.end())
			return SnapshotAcceptResult::Accepted;
		std::vector<std::uint8_t> joined;
		for (const auto& records : snapshot_transaction.records)
			joined.insert(joined.end(), records.begin(), records.end());
		phase2_evidence_result.snapshot_reassembly_complete = true;
		phase2_evidence_result.snapshot_part_count =
			snapshot_transaction.part_count;
		phase2_evidence_result.snapshot_record_count =
			snapshot_transaction.record_count;
		phase2_evidence_result.snapshot_transaction_size =
			snapshot_transaction.transaction_size;
		phase2_evidence_result.snapshot_reassembled_size =
			static_cast<std::uint32_t>(joined.size());
		protocol::Sha256Digest joined_sha{};
		phase2_evidence_result.snapshot_hash_matches =
			protocol::sha256({joined.data(), joined.size()},
				joined_sha) &&
			joined_sha ==
				snapshot_transaction.transaction_sha256;
		std::ostringstream snapshot_hash;
		for (const auto byte : joined_sha)
			snapshot_hash << std::hex << std::setfill('0')
				<< std::setw(2)
				<< static_cast<unsigned>(byte);
		phase2_evidence_result.snapshot_sha256 =
			snapshot_hash.str();
		if (joined.size() !=
				snapshot_transaction.transaction_size ||
			!phase2_evidence_result.snapshot_hash_matches)
			return SnapshotAcceptResult::Rejected;
		// A delayed snapshot from the previous manifest generation can finish
		// reassembly after the newer manifest has been installed. Classify it
		// by snapshot ordering before validating its business records against
		// the current catalog; otherwise ordinary reordering is misreported as
		// a MissingManifest corruption.
		if (client.active_snapshot_id() != 0U &&
			part.snapshot_id <= client.active_snapshot_id()) {
			snapshot_transaction.clear();
			return part.snapshot_id == client.active_snapshot_id()
				? SnapshotAcceptResult::AlreadyCommitted
				: SnapshotAcceptResult::IgnoredStale;
		}
		protocol::BusinessStateValidationContext context{};
		context.protocol_minor = protocol::VersionMinorV1_1;
		context.required_manifest_id =
			part.required_manifest_id;
		context.class_manifest_installed =
			client.installed_manifest_id() ==
			part.required_manifest_id;
		context.weapon_manifest_installed =
			context.class_manifest_installed;
		context.class_catalog = class_catalog.data();
		context.class_catalog_count = class_catalog.size();
		context.weapon_class_ids = weapon_class_ids.data();
		context.weapon_class_flags =
			weapon_class_flags.data();
		context.weapon_class_count = weapon_class_ids.size();
		protocol::BusinessStateImageValidator validator(context);
		protocol::StateImage image;
		const auto decoded =
			protocol::decode_business_snapshot_region_validated(
				{joined.data(), joined.size()},
				snapshot_transaction.record_count,
				validator, image);
		phase2_evidence_result.snapshot_decode_status =
			static_cast<std::uint32_t>(decoded);
		if (decoded != protocol::ValidationError::None) {
			std::cerr << "phase2 snapshot decode status="
				<< static_cast<unsigned>(decoded)
				<< " parts=" << snapshot_transaction.part_count
				<< " records="
				<< snapshot_transaction.record_count
				<< " bytes=" << joined.size() << "/"
				<< snapshot_transaction.transaction_size
				<< " hash="
				<< phase2_evidence_result.snapshot_sha256
				<< "\n";
			return SnapshotAcceptResult::Rejected;
		}
		const auto candidate = client.note_snapshot_candidate(part.snapshot_id,
			part.required_manifest_id, now_us,
			now_us + static_cast<std::uint64_t>(protocol::TransactionAssemblyTimeoutMs) * 1000U);
		phase2_evidence_result.snapshot_candidate_status =
			static_cast<std::uint32_t>(candidate);
		if (candidate == protocol::SnapshotCandidateResult::AlreadyCommitted) {
			snapshot_transaction.clear();
			return SnapshotAcceptResult::AlreadyCommitted;
		}
		if (candidate == protocol::SnapshotCandidateResult::Stale) {
			snapshot_transaction.clear();
			return SnapshotAcceptResult::IgnoredStale;
		}
		if (candidate != protocol::SnapshotCandidateResult::Known &&
			candidate != protocol::SnapshotCandidateResult::AlreadyKnown)
			return SnapshotAcceptResult::Rejected;
		protocol::ClientResyncChannel channel{session_id, {},
			limiter, client_resync};
		const auto committed = client.commit_snapshot(part.snapshot_id,
			part.required_manifest_id, image, now_us, channel,
			&validator);
		phase2_evidence_result.snapshot_commit_status =
			static_cast<std::uint32_t>(committed);
		if (committed != protocol::SnapshotCommitResult::Committed &&
			committed != protocol::SnapshotCommitResult::CommittedAndPendingDeltaApplied &&
			committed != protocol::SnapshotCommitResult::AlreadyCommitted)
			return SnapshotAcceptResult::Rejected;
		client_published_us = now_us;
		++client_publications;
		phase2_evidence_result.client_published = !client.published().empty();
		snapshot_transaction.clear();
		return SnapshotAcceptResult::Accepted;
	}
#endif
	bool consume(Backend& backend, std::uint64_t now_us,
		std::uint64_t active_producer_session_id = 0U) noexcept {
#if defined(FSO_TELEMETRY_PHASE2_EVIDENCE)
		backend.phase2_flush_downstream();
#endif
		while(tx_seen<backend.tx.size()) {
			const auto& packet=backend.tx[tx_seen++]; protocol::DatagramView d{};
			const auto decoded=protocol::decode_and_validate_datagram(bytes(packet.data),{protocol::VersionMinorV1_1,protocol::VersionMinorV1_1},d);
			if(decoded!=protocol::ValidationError::None) { std::cerr<<"peer packet validation="<<static_cast<unsigned>(decoded)<<" tx="<<(tx_seen-1U)<<" bytes="<<packet.data.size()<<"\n"; return false; }
			if (d.header.message_type !=
					protocol::MessageType::Welcome &&
				((active_producer_session_id != 0U &&
				  d.header.session_id !=
					  active_producer_session_id) ||
				 session_id == 0U ||
				 d.header.session_id != session_id)) {
				// Delayed output from a terminal session is outside the
				// independent client's current authority. Ignore it before it
				// can mutate ACK, reassembly, snapshot or resync state.
				continue;
			}
			if(d.header.message_type==protocol::MessageType::Welcome) {
				if (active_producer_session_id != 0U &&
					d.header.session_id != active_producer_session_id) {
					continue;
				}
				if(session_id!=0U && session_id!=d.header.session_id) {
					std::cerr<<"peer welcome session changed old="<<session_id<<" new="<<d.header.session_id<<"\n";
					begin_new_session(d.header.session_id);
				} else {
					session_id=d.header.session_id;
				}
				Backend::Packet ack{}; if(!make_ack(packet,sequence++,ack)){ std::cerr<<"peer welcome ACK encode failed\n"; return false; } backend.enqueue_peer(std::move(ack)); welcome=true;
			}
			else if(d.header.message_type==protocol::MessageType::SessionBegin) { Backend::Packet ack{}; if(!make_ack(packet,sequence++,ack))return false; backend.enqueue_peer(std::move(ack)); begin=true; }
#if defined(FSO_TELEMETRY_PHASE2_EVIDENCE)
			else if(d.header.message_type==protocol::MessageType::Manifest) {
				protocol::ReassembledMessage completed{};
				auto result = reassembler.ingest(d, completed);
				if (result ==
					protocol::ReassemblyResult::QuotaExceeded) {
					reassembler.clear();
					result = reassembler.ingest(d, completed);
				}
				if (result == protocol::ReassemblyResult::Accepted ||
					result == protocol::ReassemblyResult::Duplicate)
					continue;
				if (result != protocol::ReassemblyResult::Completed) {
					std::cerr << "phase2 peer manifest reassembly failed status="
							  << static_cast<unsigned>(result)
							  << " message=" << d.header.message_id << "\n";
					return false;
				}
				const protocol::DatagramView logical{completed.header, completed.payload_view()};
				if (!accept_manifest(packet, logical, now_us, backend)) {
					std::cerr << "phase2 peer manifest accept failed message="
							  << logical.header.message_id << "\n";
					return false;
				}
			}
#endif
			else if(d.header.message_type==protocol::MessageType::FullSnapshot) {
				protocol::ReassembledMessage completed{};
				auto result = reassembler.ingest(d, completed);
				if (result ==
					protocol::ReassemblyResult::QuotaExceeded) {
					reassembler.clear();
					result = reassembler.ingest(d, completed);
				}
				if (result == protocol::ReassemblyResult::Accepted ||
					result == protocol::ReassemblyResult::Duplicate)
					continue;
				if (result != protocol::ReassemblyResult::Completed) {
					std::cerr << "phase2 peer snapshot reassembly failed status="
							  << static_cast<unsigned>(result)
							  << " message=" << d.header.message_id << "\n";
					return false;
				}
				const protocol::DatagramView logical{completed.header, completed.payload_view()};
				protocol::FullSnapshotPartPayload snapshot_part{};
				if(protocol::decode_full_snapshot_part_payload(logical.payload,snapshot_part)!=protocol::ValidationError::None) return false;
#if defined(FSO_TELEMETRY_PHASE2_EVIDENCE)
				const auto accepted_snapshot =
					accept_snapshot_part(snapshot_part, logical, now_us);
				if (accepted_snapshot == SnapshotAcceptResult::Rejected) {
					std::cerr << "phase2 peer snapshot accept rejected message="
							  << logical.header.message_id << "\n";
					return false;
				}
				if (accepted_snapshot == SnapshotAcceptResult::IgnoredStale) {
					reassembler.clear();
					continue;
				}
				if (accepted_snapshot == SnapshotAcceptResult::AlreadyCommitted) {
					if (resync_sent &&
						snapshot_part.snapshot_flags ==
							protocol::SnapshotFlagResync) {
						if (snapshot_part.snapshot_id ==
							initial_snapshot_id)
							return false;
						resync_snapshot_flag_seen = true;
						recovered_snapshot = true;
						slow_client_recovered = true;
						recovery_snapshot_us = now_us;
					}
					Backend::Packet ack{};
					if (!make_ack(packet, sequence++, ack)) return false;
					backend.enqueue_peer(std::move(ack));
					if (pending.session == logical.header.session_id &&
						pending.id == logical.header.message_id &&
						pending.crc == logical.header.message_crc32 &&
						pending.count == logical.header.fragment_count)
						pending.acked = true;
					continue;
				}
				if (client.active_snapshot_id() ==
					snapshot_part.snapshot_id)
					reassembler.clear();
#endif
				// A periodic candidate already queued before the accepted resync may
				// legitimately drain first.  It is not the replacement proof; require
				// instead that a subsequent fresh RESYNC candidate is observed.
				if(resync_sent && snapshot_part.snapshot_flags==protocol::SnapshotFlagResync) { if(snapshot_part.snapshot_id==initial_snapshot_id) return false; resync_snapshot_flag_seen=true; }
				if(!pending.count) { pending={logical.header.session_id,logical.header.message_id,logical.header.message_crc32,logical.header.fragment_count,false}; }
				if(pending.session!=logical.header.session_id || pending.id!=logical.header.message_id || pending.crc!=logical.header.message_crc32 || pending.count!=logical.header.fragment_count) {
					// A reliable candidate may reach terminal policy under the
					// measured impairment and be replaced by a fresh cumulative
					// keyframe. The incomplete candidate was never published;
					// discard its ACK bookkeeping and follow the new transaction.
					pending={logical.header.session_id,logical.header.message_id,logical.header.message_crc32,logical.header.fragment_count,false};
				}
				if(!pending.acked) {
					Backend::Packet ack{}; if(!make_ack(packet,sequence++,ack))return false;
					if(!snapshot) { delayed_snapshot_ack=ack; backend.enqueue_duplicate(std::move(ack)); duplicate_delivered=true; snapshot=true; }
					else if(resync_sent) {
						if(withhold_recovery_ack) { withhold_recovery_ack=false; ++ack_loss; ++slow_client_withheld_reliable_acks; continue; }
						if(!reorder_delivered) { if(delayed_snapshot_ack.data.empty()) return false; backend.enqueue_reordered(std::move(ack),std::move(delayed_snapshot_ack)); reorder_delivered=true; }
						else backend.enqueue_peer(std::move(ack));
						recovered_snapshot=true; slow_client_recovered=true; recovery_snapshot_us=now_us;
					} else backend.enqueue_peer(std::move(ack));
					pending.acked=true;
				}
			}
			else if(d.header.message_type==protocol::MessageType::Delta && snapshot) {
#if defined(FSO_TELEMETRY_PHASE2_EVIDENCE)
				protocol::ReassembledMessage completed{};
				auto reassembly = reassembler.ingest(d, completed);
				// Loss plus bounded reordering can leave the client's bounded
				// delta slots occupied by superseded cumulative messages. Drop
				// those incomplete candidates and admit the newest fragment;
				// no prior delta is required for eventual convergence.
				if (reassembly ==
					protocol::ReassemblyResult::QuotaExceeded) {
					reassembler.clear();
					reassembly = reassembler.ingest(d, completed);
				}
				if (reassembly == protocol::ReassemblyResult::Accepted ||
					reassembly == protocol::ReassemblyResult::Duplicate)
					continue;
				if (reassembly != protocol::ReassemblyResult::Completed) {
					std::cerr << "phase2 peer delta reassembly failed status="
							  << static_cast<unsigned>(reassembly)
							  << " message=" << d.header.message_id << "\n";
					return false;
				}
				const protocol::DatagramView logical{
					completed.header, completed.payload_view()};
				protocol::DeltaPayload payload{};
				protocol::CumulativeStateDelta decoded_delta{};
				const auto delta_payload_status =
					protocol::decode_delta_payload(logical.payload, payload);
				const auto business_delta_status =
					delta_payload_status == protocol::ValidationError::None
					? protocol::decode_business_delta(
						payload, logical.header.version_minor, decoded_delta)
					: protocol::ValidationError::TruncatedPayload;
				if (delta_payload_status != protocol::ValidationError::None ||
					business_delta_status != protocol::ValidationError::None) {
					std::cerr << "phase2 peer delta decode failed payload="
							  << static_cast<unsigned>(delta_payload_status)
							  << " business="
							  << static_cast<unsigned>(business_delta_status)
							  << " message="
							  << logical.header.message_id << "\n";
					return false;
				}
				const auto applied = client.receive_delta(decoded_delta, now_us, session_id,
					packet.endpoint, limiter, client_resync);
				if (protocol::client_delta_validation_error(applied) != protocol::ValidationError::None &&
					applied != protocol::ClientDeltaResult::UnknownBaselineResyncRequested &&
					applied != protocol::ClientDeltaResult::UnknownBaselineRateLimited) {
					std::cerr << "phase2 peer delta apply failed status="
							  << static_cast<unsigned>(applied)
							  << " base=" << decoded_delta.baseline_snapshot_id
							  << " active=" << client.active_snapshot_id()
							  << " message="
							  << logical.header.message_id << "\n";
					return false;
				}
				if (applied == protocol::ClientDeltaResult::Applied) {
					client_published_us = now_us;
					++client_publications;
				}
#endif
				delta_observed=true; if(!resync_sent && now_us>=next_resync_us) { Backend::Packet request{}; const auto request_sequence=sequence++; if(!make_resync(packet,request_sequence,request))return false; resync_message_id=request_sequence; resync_packet=request; backend.enqueue_peer(std::move(request)); delta=true; resync_sent=true; resync_started_us=now_us; next_resync_retry_us=now_us+250000U; resync_retries=0U; resync_ack_us=0U; recovery_snapshot_us=0U; ++resync_cycles; pending={}; }
			}
			else if(d.header.message_type==protocol::MessageType::Heartbeat) {
				protocol::HeartbeatPayload heartbeat{};
				if (protocol::decode_heartbeat_payload(
						d.payload, heartbeat) !=
						protocol::ValidationError::None)
					return false;
				if (heartbeat.kind ==
					protocol::HeartbeatKind::Response) {
#if defined(FSO_TELEMETRY_PHASE2_EVIDENCE)
					const auto correlated = std::find_if(
						blackout_heartbeat_probes.begin(),
						blackout_heartbeat_probes.end(),
						[&](const auto& probe) {
							return probe.session_id ==
									d.header.session_id &&
								probe.probe_id ==
									heartbeat.probe_id &&
								probe.origin_t0_us ==
									heartbeat.origin_t0_us;
						});
					const bool timestamps_valid =
						heartbeat.origin_t0_us <=
							heartbeat.receive_t1_us &&
						heartbeat.receive_t1_us <=
							heartbeat.transmit_t2_us &&
						heartbeat.transmit_t2_us <=
							d.header.sent_time_us &&
						d.header.sent_time_us <= now_us;
					if (d.header.session_id != session_id)
						return false;
					if (correlated ==
						blackout_heartbeat_probes.end())
						continue;
					if (!timestamps_valid)
						continue;
					blackout_heartbeat_probes.erase(correlated);
					continue;
#else
					return false;
#endif
				}
				if (heartbeat.kind !=
					protocol::HeartbeatKind::Request)
					return false;
				const bool arm_post_resync_heartbeat =
					resync_ack &&
					!post_resync_heartbeat_armed &&
					d.header.session_id == session_id;
				if (arm_post_resync_heartbeat) {
					post_resync_heartbeat_session_id =
						d.header.session_id;
					post_resync_heartbeat_probe_id =
						heartbeat.probe_id;
					post_resync_heartbeat_origin_t0_us =
						heartbeat.origin_t0_us;
					post_resync_heartbeat_armed = true;
				}
				Backend::Packet response{};
				if(!make_heartbeat_response(
						packet,sequence++,now_us,response))
					return false;
				backend.enqueue_peer(std::move(response));
				if (arm_post_resync_heartbeat)
					post_resync_heartbeat_response_sent =
						true;
			}
			else if(d.header.message_type==protocol::MessageType::SessionEnd) { protocol::SessionEndPayload end{}; if(protocol::decode_session_end_payload(d.payload,end)!=protocol::ValidationError::None)return false; session_end_seen=true; session_end_reason=static_cast<std::uint8_t>(end.reason); }
			else if(d.header.message_type==protocol::MessageType::Ack && resync_sent) { protocol::AckPayload ack{}; if(protocol::decode_ack_payload(d.payload,ack)!=protocol::ValidationError::None){ std::cerr<<"phase2 peer resync ACK decode failed\n"; return false; } if(ack.target_message_type==protocol::MessageType::ResyncRequest && ack.target_message_id==resync_message_id && ack.target_fragment_count==1U) { resync_ack=true; resync_ack_us=now_us; } else { std::cerr<<"phase2 peer unexpected ACK target-type="<<static_cast<unsigned>(ack.target_message_type)<<" target-id="<<ack.target_message_id<<" expected-id="<<resync_message_id<<"\n"; invalid_ack=true; } }
		}
		return true;
	}
	bool retry_resync(Backend& backend, std::uint64_t now_us) noexcept {
		if(!resync_sent || resync_ack || now_us<next_resync_retry_us || resync_packet.data.empty()) return true;
		if(resync_retries>=8U) return true;
		Backend::Packet retry{}; if(!retransmit_reliable(resync_packet,sequence++,now_us,retry)) return false;
		backend.enqueue_peer(std::move(retry)); ++resync_retries; next_resync_retry_us=now_us+(250000U<<std::min<std::uint32_t>(resync_retries,4U)); return true;
	}
	bool request_final_resync(Backend& backend,
		const protocol::EndpointKey& endpoint,
		std::uint64_t now_us,
		bool transport_drained,
		bool& requested) noexcept {
		requested = false;
		if (!phase2_final_resync_handoff_ready(
				resync_sent, transport_drained))
			return true;
		if (session_id == 0U) return false;
		Backend::Packet request{};
		const auto request_sequence = sequence++;
		if (!make_resync_for_session(session_id, endpoint,
				request_sequence, now_us, request))
			return false;
		resync_message_id = request_sequence;
		resync_packet = request;
		backend.enqueue_peer(std::move(request));
		resync_sent = true;
		resync_ack = false;
		recovered_snapshot = false;
		recovery_loss_armed = true;
		resync_started_us = now_us;
		next_resync_retry_us = now_us + 250000U;
		resync_retries = 0U;
		resync_ack_us = 0U;
		recovery_snapshot_us = 0U;
		pending = {};
#if defined(FSO_TELEMETRY_PHASE2_EVIDENCE)
		post_resync_heartbeat_session_id = 0U;
		post_resync_heartbeat_origin_t0_us = 0U;
		post_resync_heartbeat_probe_id = 0U;
		post_resync_heartbeat_armed = false;
		post_resync_heartbeat_response_sent = false;
#endif
		requested = true;
		return true;
	}
	bool close_completed_resync_cycle(
		const detail::SessionControllerSlot& slot,
		std::uint64_t now_us) noexcept {
		if (!phase2_resync_cycle_promoted(
				resync_sent, recovered_snapshot,
				initial_snapshot_id,
				slot.snapshot.active_snapshot_id()))
			return false;
		recovered_baseline_published = true;
		++successful_recoveries;
		const auto convergence = now_us - resync_started_us;
		if (convergence > max_convergence_us) {
			max_convergence_us = convergence;
			worst_resync_start_us = resync_started_us;
			worst_resync_ack_us = resync_ack_us;
			worst_snapshot_us = recovery_snapshot_us;
			worst_live_us = now_us;
		}
		initial_snapshot_id = slot.snapshot.active_snapshot_id();
		resync_sent = false;
		resync_ack = false;
		recovered_snapshot = false;
		next_resync_us = now_us + 5000000U;
		return true;
	}
	void abandon_terminal_resync_cycle() noexcept {
		resync_message_id = 0U;
		resync_packet = {};
		resync_sent = false;
		resync_ack = false;
		recovered_snapshot = false;
		recovery_loss_armed = false;
		next_resync_retry_us = 0U;
		resync_retries = 0U;
		resync_started_us = 0U;
		resync_ack_us = 0U;
		recovery_snapshot_us = 0U;
		pending = {};
#if defined(FSO_TELEMETRY_PHASE2_EVIDENCE)
		post_resync_heartbeat_session_id = 0U;
		post_resync_heartbeat_origin_t0_us = 0U;
		post_resync_heartbeat_probe_id = 0U;
		post_resync_heartbeat_armed = false;
		post_resync_heartbeat_response_sent = false;
#endif
	}
	bool emit_blackout_probe(
		Backend& backend, const protocol::EndpointKey& endpoint,
		std::uint64_t active_session_id,
		std::uint64_t now_us, std::size_t window,
		bool lead_in) noexcept {
		Backend::Packet request{};
		if (active_session_id == 0U ||
			!make_heartbeat_request(
				active_session_id, endpoint, sequence, now_us,
				request)) {
			return false;
		}
#if defined(FSO_TELEMETRY_PHASE2_EVIDENCE)
		if (blackout_heartbeat_probes.size() >=
			protocol::MaxInFlightProbes)
			blackout_heartbeat_probes.erase(
				blackout_heartbeat_probes.begin());
		blackout_heartbeat_probes.push_back(
			{active_session_id, sequence, now_us});
		phase2_evidence_probes.push_back(
			{active_session_id, sequence, now_us, window,
			 lead_in});
		auto& last_emission =
			phase2_evidence_result
				.blackout_probe_last_emission_us[window];
		if (last_emission != 0U) {
			const auto spacing = now_us - last_emission;
			auto& minimum =
				phase2_evidence_result
					.blackout_probe_minimum_spacing_us[window];
			minimum = minimum == 0U
				? spacing
				: std::min(minimum, spacing);
		}
		last_emission = now_us;
		++phase2_evidence_result
			  .blackout_probe_emissions[window];
#endif
		++sequence;
		backend.enqueue_peer(std::move(request));
		return true;
	}
};

// Reuses the real native runtime, UDP backend, wire peer, and snapshot ACK
// path after a lifecycle discontinuity. It is intentionally test-only and
// proves a newly negotiated live baseline rather than merely reconstructing
// a local fixture state.
bool converge_after_discontinuity(detail::NativeSessionRuntime& runtime, Backend& backend, const Engine& engine,
	Peer& peer, const protocol::EndpointKey& endpoint, std::uint64_t& now, std::uint32_t mission_generation) noexcept
{
	backend.reset_peer_io();
	Backend::Packet hello{};
	if (!make_hello(endpoint,hello)) return false;
	backend.enqueue_peer(std::move(hello));
	for (std::uint64_t index=0U; index<256U; ++index) {
		backend.now_us=now;
		if (index<32U ? !r2_tick(runtime,now,mission_generation) : !tick(runtime,engine,now,mission_generation)) return false;
		const auto* slot=detail::NativeSessionRuntimeTestAccess::slot(runtime,0U);
		if (!peer.consume(backend,now,
				slot == nullptr ? 0U : slot->session_id) ||
			!peer.retry_resync(backend,now)) return false;
		if (slot!=nullptr && slot->snapshot.has_active_baseline()) {
			peer.live_baseline_published=true;
			return peer.welcome && peer.begin && peer.snapshot && peer.session_id!=0U;
		}
		now+=33334U;
	}
	return false;
}

				 terminal_resync_output_bypass);
			phase2_evidence_result.terminal_output_bypass =
				terminal_resync_output_bypass;
			phase2_evidence_result.peer_resync_sent_at_drain =
				convergence_peer->resync_sent;
			phase2_evidence_result.final_resync_prerequisites =
				phase2_evidence_result.t0_us == 0U &&
				observed_slot != nullptr &&
				observed_slot->snapshot.has_active_baseline() &&
				latest_manifest_applied &&
				final_resync_session_id !=
					observed_slot->session_id;
		};
		if (phase2_evidence_result.t0_us == 0U &&
			final_resync_session_id != 0U &&
			(observed_slot == nullptr ||
			 convergence_peer->session_id !=
				 final_resync_session_id ||
			 observed_slot->session_id !=
				 final_resync_session_id)) {
			final_resync_session_id = 0U;
			final_resync_ack_tick =
				std::numeric_limits<std::uint64_t>::max();
			final_resync_ack_clock_response_us = 0U;
		}
		if (phase2_evidence_result.t0_us == 0U &&
			observed_slot != nullptr &&
			observed_slot->snapshot.has_active_baseline() &&
			final_resync_session_id != observed_slot->session_id) {
			bool requested = false;
			if (!convergence_peer->request_final_resync(
					backend, endpoint, now,
					impairment_drained, requested)) {
				std::cerr << "phase2 final resync encode failed\n";
				runtime->shutdown();
				return 4;
			}
			if (requested) {
				++phase2_evidence_result
					  .final_resync_request_count;
				const auto arm_index =
					phase2_evidence_result
						.final_resync_arm_timestamp_count;
				if (arm_index <
					phase2_evidence_result
						.final_resync_arm_us.size()) {
					phase2_evidence_result
						.final_resync_arm_us[arm_index] = now;
					phase2_evidence_result
						.final_resync_progress_at_arm[arm_index] =
						static_cast<std::uint8_t>(
							observed_slot->progress);
					phase2_evidence_result
						.final_resync_message_id_at_arm[arm_index] =
						convergence_peer->resync_message_id;
					++phase2_evidence_result
						  .final_resync_arm_timestamp_count;
				}
				recovery_deadline_us = std::max(
					recovery_deadline_us,
					phase2_final_resync_attempt_deadline(now));
				phase2_evidence_result.recovery_deadline_us =
					recovery_deadline_us;
				terminal_resync_output_bypass = false;
				final_resync_session_id = observed_slot->session_id;
				final_resync_ack_tick =
					std::numeric_limits<std::uint64_t>::max();
				final_resync_ack_clock_response_us = 0U;
				post_resync_heartbeat_budget_reserved = false;
			} else
				++phase2_evidence_result
					  .final_resync_suppressed_count;
		}
		if (phase2_evidence_result.t0_us == 0U &&
			observed_slot != nullptr &&
			final_resync_session_id == observed_slot->session_id &&
			convergence_peer->session_id ==
				final_resync_session_id &&
			convergence_peer->resync_ack &&
			final_resync_ack_tick ==
				std::numeric_limits<std::uint64_t>::max()) {
			final_resync_ack_tick = recovery_tick;
			final_resync_ack_clock_response_us =
				observed_slot->heartbeat
					.last_valid_clock_response_us;
		}
		// The reliable retry envelope ends when the replacement snapshot is
		// published. Reserve exactly one non-sliding heartbeat period after
		// that publication so the independent liveness barrier can complete.
		if (!post_resync_heartbeat_budget_reserved &&
			phase2_evidence_result.t0_us == 0U &&
			observed_slot != nullptr &&
			final_resync_session_id == observed_slot->session_id &&
			convergence_peer->resync_ack &&
			convergence_peer->recovered_snapshot) {
			recovery_deadline_us = std::max(
				recovery_deadline_us,
				phase2_post_resync_heartbeat_deadline(
					now,
					observed_slot->heartbeat
						.negotiated_interval_ms));
			phase2_evidence_result.recovery_deadline_us =
				recovery_deadline_us;
			post_resync_heartbeat_budget_reserved = true;
		}
		// The peer can observe the first post-ACK heartbeat request on the
		// final turn of the period reserved above. Its response is then queued
		// for producer ingress on the following turn. Reserve that delivery
		// exactly once; this is a bounded transport handoff, not a sliding
		// retry extension.
		if (!post_resync_heartbeat_delivery_budget_reserved &&
			phase2_evidence_result.t0_us == 0U &&
			observed_slot != nullptr &&
			convergence_peer->post_resync_heartbeat_response_sent &&
			convergence_peer->
					post_resync_heartbeat_origin_t0_us != 0U &&
			observed_slot->heartbeat.last_valid_clock_response_us <
				convergence_peer->
					post_resync_heartbeat_origin_t0_us) {
			recovery_deadline_us = std::max(
				recovery_deadline_us,
				phase2_saturating_deadline(
					now,
					Phase2MaximumDeliveryDelayUs +
						4U * Phase2RecoveryTickUs));
			phase2_evidence_result.recovery_deadline_us =
				recovery_deadline_us;
			post_resync_heartbeat_delivery_budget_reserved = true;
		}

		// following scheduler turn may arm a reliable replacement snapshot.
		// Liveness is instead proven by the first Heartbeat Request observed
		// after that exact ACK: the peer latches its session/probe/origin tuple,
		// sends the correlated response, and the production clock-response
		// timestamp must then advance on the same Ready slot. Reliable snapshot
		// work remains part of convergence/resource proof and is deliberately
		// not consumed as a prerequisite for t0.
		const bool heartbeat_response_correlated =
			observed_slot != nullptr &&
			convergence_peer->post_resync_heartbeat_armed &&
			convergence_peer->
				post_resync_heartbeat_response_sent &&
			convergence_peer->
				post_resync_heartbeat_session_id ==
				observed_slot->session_id &&
			convergence_peer->
				post_resync_heartbeat_probe_id != 0U &&
			convergence_peer->
				post_resync_heartbeat_origin_t0_us != 0U &&
			observed_slot->heartbeat
					.last_valid_clock_response_us >=
				final_resync_ack_clock_response_us &&
			observed_slot->heartbeat
					.last_valid_clock_response_us >=
				convergence_peer->
					post_resync_heartbeat_origin_t0_us;

			observed_slot != nullptr &&
			observed_slot->progress ==
				detail::ProducerSessionProgress::ReadyForState &&
			!observed_slot->heartbeat.clock_stale &&
			observed_slot->heartbeat.last_valid_network_activity_us !=
				0U &&
			observed_slot->heartbeat.disconnect_timeout_us != 0U &&

					   .last_valid_network_activity_us &&
			now - observed_slot->heartbeat
					  .last_valid_network_activity_us <
				observed_slot->heartbeat.disconnect_timeout_us;

			observed_slot != nullptr &&
			final_resync_session_id == observed_slot->session_id &&
			convergence_peer->session_id ==
				final_resync_session_id &&
			final_resync_ack_tick !=
				std::numeric_limits<std::uint64_t>::max() &&
			heartbeat_response_correlated &&
			session_clock_valid &&
			observed_slot->required_manifest_applied &&
			!observed_slot->fault_session_end_pending &&
			latest_manifest_applied &&
			impairment_drained;

			heartbeat_response_correlated;
		last_session_clock_valid = session_clock_valid;
		last_latest_manifest_applied = latest_manifest_applied;
		last_impairment_drained = impairment_drained;
		last_required_manifest_applied =
			observed_slot != nullptr &&
			observed_slot->required_manifest_applied;

			observed_slot != nullptr &&
			observed_slot->fault_session_end_pending;

				return 4;

					phase2_evidence_result
						.final_resync_arm_timestamp_count;

					phase2_evidence_result

						.final_resync_message_id_at_arm[arm_index] =
						convergence_peer->resync_message_id;
					++phase2_evidence_result

					convergence_peer->client_publications;
				replacement_snapshot_id_at_request =
					convergence_peer->client.active_snapshot_id();
				convergence_peer->resync_snapshot_flag_seen = false;
				post_t0_resync_sent = true;
				phase2_evidence_result.replacement_request_us = now;
			} else
				++phase2_evidence_result
					  .final_resync_suppressed_count;
		}

		if (convergence_peer->client.active_snapshot_id() != 0U) {

				baseline_history.begin(),
				baseline_history.begin() +
					static_cast<std::ptrdiff_t>(
						baseline_history_size),
				[&](const auto& entry) {
					return entry.snapshot_id ==
							convergence_peer->client.active_snapshot_id() &&
						entry.baseline ==
							convergence_peer->client.active_baseline();
				});
			last_client_provenance =
				found != baseline_history.begin() +
					static_cast<std::ptrdiff_t>(
						baseline_history_size);
		}
		const bool replacement_snapshot_live =
			phase2_replacement_snapshot_is_live(
				post_t0_resync_sent,
				convergence_peer->resync_ack,
				convergence_peer->resync_snapshot_flag_seen,
				convergence_peer->recovered_snapshot,
				convergence_peer->client_published_us,
				phase2_evidence_result.t0_us,
				phase2_evidence_result.replacement_request_us,
				convergence_peer->client_publications,
				client_publications_at_t0,
				replacement_publications_at_request,
				convergence_peer->client.active_snapshot_id(),
				snapshot_id_at_t0,
				replacement_snapshot_id_at_request);
		if (oracle_frozen && replacement_snapshot_live &&
			latest_manifest_applied &&
			last_client_provenance &&
			!convergence_peer->client.published().empty() &&
			convergence_peer->client.published() == oracle_stable) {
			phase2_evidence_result.client_equals_producer = true;
			phase2_evidence_result.live_after_replacement_snapshot =
				true;
			phase2_evidence_result.replacement_snapshot_applied_us =
				convergence_peer->client_published_us;
			phase2_evidence_result.replacement_snapshot_id =
				convergence_peer->client.active_snapshot_id();
			phase2_evidence_result.converged_us = now;
			phase2_evidence_result.converged_peer_session_id =
				convergence_peer->session_id;
			phase2_evidence_result.oracle_state_sha256 =
				oracle_stable_sha256;
			phase2_evidence_result.client_state_sha256 =
				phase2_state_sha256(convergence_peer->client.published());
			phase2_evidence_result.recovery_exit_reason =
				Phase2RecoveryExitReason::Converged;
			phase2_evidence_result.recovery_turns =
				recovery_tick + 1U;
			phase2_evidence_result.recovery_retry_count_at_exit =
				convergence_peer->resync_retries;
			phase2_evidence_result
				.recovery_next_retry_us_at_exit =
				convergence_peer->next_resync_retry_us;
			capture_drain_diagnostics();
			break;
		}
		capture_drain_diagnostics();
		phase2_evidence_result.recovery_turns =
			recovery_tick + 1U;
		std::uint64_t next_recovery_us = now;
		if (!phase2_advance_recovery_clock(
				now, recovery_deadline_us,
				next_recovery_us)) {
			phase2_evidence_result.recovery_exit_reason =
				Phase2RecoveryExitReason::DeadlineExpired;
			break;
		}
		now = next_recovery_us;
	}
	if (phase2_evidence_result.recovery_exit_reason ==
		Phase2RecoveryExitReason::None) {
		phase2_evidence_result.recovery_exit_reason =
			Phase2RecoveryExitReason::DeadlineExpired;
	}
	phase2_evidence_result.recovery_deadline_us =
		recovery_deadline_us;
	phase2_evidence_result.recovery_retry_count_at_exit =
		convergence_peer->resync_retries;
	phase2_evidence_result.recovery_next_retry_us_at_exit =
		convergence_peer->next_resync_retry_us;
	if (phase2_evidence_result.converged_us == 0U) {

				  << (convergence_peer->client.published() ==

					  << (diagnostic_slot->delta_egress.has_delta()
							? 1U : 0U)
					  << " heartbeat-interval-ms="

							 .last_valid_clock_response_us

		std::cerr << " history=";
		for (std::size_t index = 0U;
			 index < baseline_history_size; ++index) {

					  << ":"
					  << phase2_state_sha256(
							 baseline_history[index].baseline);
		}
		std::cerr << " drain-diagnostic="
				  << "abandoned:"
				  << phase2_evidence_result

						 .final_resync_request_count
				  << ",suppressed:"
				  << phase2_evidence_result
						 .final_resync_suppressed_count
				  << ",stable:"
				  << phase2_evidence_result
						 .terminal_resync_stable_turns
				  << ",upstream:"
				  << phase2_evidence_result
						 .drain_upstream_scheduled
				  << ",downstream:"
				  << phase2_evidence_result
						 .drain_downstream_scheduled
				  << ",rx:"
				  << phase2_evidence_result.drain_rx_remaining
				  << ",tx:"
				  << phase2_evidence_result.drain_tx_unseen

				  << (convergence_peer->

				  << phase2_evidence_result
						 .final_resync_arm_timestamp_count
				  << ",arms:"
				  << phase2_evidence_result.final_resync_arm_us[0]
				  << "/"
				  << phase2_evidence_result.final_resync_arm_us[1]
				  << "/"
				  << phase2_evidence_result.final_resync_arm_us[2]

				  << phase2_evidence_result

		lifecycle.mission_generation_before=1U; lifecycle.mission_generation_after=2U;
		runtime->purge_all(detail::SessionCloseReason::MissionDiscontinuity);
		Peer mission_peer{};
		lifecycle.mission_converged=converge_after_discontinuity(*runtime,backend,engine,mission_peer,endpoint,now,2U) && mission_peer.session_id!=peer.session_id;
		lifecycle.mission_session_id=mission_peer.session_id;
	}
	runtime->shutdown();
#if defined(FSO_TELEMETRY_PHASE2_EVIDENCE)
	phase2_evidence_result.resources_bounded =
		runtime->socket_count() == 0U &&
		runtime->owned_usage() == detail::SessionControllerOwnedUsage{};
	if (!convergence_peer->client.published().empty()) {
		bool lifecycle = false;
		bool support = phase2_evidence_profile == telemetry::Phase2Profile::CoreGate;
		for (const auto& atom : convergence_peer->client.published().records()) {
			lifecycle = lifecycle || atom.key.record_type ==
				static_cast<std::uint16_t>(protocol::RecordType::EntityLifecycle);
			support = support || atom.key.record_type ==
				static_cast<std::uint16_t>(protocol::RecordType::SupportState);
		}
		phase2_evidence_result.terminal_exact = lifecycle && support;
	}
#endif
	lifecycle.normal_shutdown_released=runtime->socket_count()==0U && runtime->owned_usage()==detail::SessionControllerOwnedUsage{};
	if (restart_process) {
		Backend restarted_backend{}; Completion restarted_completion{}; FixedRandom restarted_ids_random{},restarted_packet_random{};
		restarted_ids_random.value=0x1000U; restarted_packet_random.value=0x2000U;
		detail::SessionIdRegistry restarted_registry; detail::SessionIdAllocator restarted_ids(restarted_ids_random,restarted_registry);
		if (restarted_registry.allocate_storage()) {
			auto restarted_runtime=std::make_unique<detail::NativeSessionRuntime>(restarted_backend,restarted_completion);
			if (restarted_runtime->start({&config,0x1020304050607080ULL,&restarted_ids,&restarted_packet_random})==detail::NativeSessionStartStatus::Started) {
				Peer process_peer{};
				lifecycle.process_converged=converge_after_discontinuity(*restarted_runtime,restarted_backend,engine,process_peer,endpoint,now,1U) && process_peer.session_id!=peer.session_id;
				lifecycle.process_session_id=process_peer.session_id;
				restarted_runtime->shutdown();
				lifecycle.process_shutdown_released=restarted_runtime->socket_count()==0U && restarted_runtime->owned_usage()==detail::SessionControllerOwnedUsage{};
			}
		}
	}
#if defined(FSO_TELEMETRY_PHASE2_EVIDENCE)
	const bool complete = convergence_peer->welcome &&
		convergence_peer->begin && convergence_peer->snapshot &&
		phase2_evidence_result.manifest_installed &&
		phase2_evidence_result.client_published &&
		phase2_evidence_result.client_equals_producer &&
		phase2_evidence_result.live_after_replacement_snapshot &&
		phase2_evidence_result.no_partial_publication &&
		phase2_evidence_result.resources_bounded &&
		phase2_evidence_result.terminal_exact &&
		phase2_evidence_result.directional_records[0] != 0U &&
		phase2_evidence_result.directional_records[1] != 0U &&
		phase2_evidence_result.converged_us >= phase2_evidence_result.t0_us &&
		phase2_evidence_result.converged_us - phase2_evidence_result.t0_us <= 5000000ULL;
#else
	const bool complete=peer.welcome&&peer.begin&&peer.snapshot&&peer.delta&&peer.resync_cycles!=0U&&peer.successful_recoveries!=0U&&peer.duplicate_delivered&&peer.reorder_delivered&&peer.resync_snapshot_flag_seen&&peer.live_baseline_published&&peer.recovered_baseline_published&&peer.slow_client_withheld_reliable_acks!=0U&&peer.slow_client_recovered&&!peer.invalid_ack&&backend.loss!=0U&&backend.delta_loss!=0U&&backend.snapshot_fragment_loss!=0U&&backend.outage_loss!=0U&&peer.ack_loss!=0U&&backend.jitter_delayed!=0U&&backend.would_block_injected!=0U&&backend.bounded_impairment_start_us!=0U&&backend.bounded_impairment_end_us>backend.bounded_impairment_start_us&&recovered_within_ten_seconds_after_end&&observations.size()==duration_seconds/60U&&final_live&&lifecycle.normal_shutdown_released&&(!restart_mission||lifecycle.mission_converged)&&(!restart_process||(lifecycle.process_converged&&lifecycle.process_shutdown_released));
#endif
	return write_report(report,profile,seed,duration,backend,*convergence_peer,observations,lifecycle,complete) ? (complete?0:4) : 5;
}
