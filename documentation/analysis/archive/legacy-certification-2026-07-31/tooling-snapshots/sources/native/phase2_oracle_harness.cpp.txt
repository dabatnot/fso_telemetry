constexpr std::uint64_t SessionId = 0x1122334455667788ULL;

template <typename Container>
ByteView bytes(const Container& value)
{
	return {reinterpret_cast<const std::uint8_t*>(value.data()), value.size()};
}
std::vector<std::uint8_t> read_binary(const fs::path& path)
{
	std::ifstream input(path, std::ios::binary);
	return {std::istreambuf_iterator<char>(input), std::istreambuf_iterator<char>()};
}
std::string read_text(const fs::path& path)
{
	std::ifstream input(path, std::ios::binary);
	return {std::istreambuf_iterator<char>(input), std::istreambuf_iterator<char>()};
}

bool write_binary(const fs::path& path, const std::vector<std::uint8_t>& value)
{
	std::ofstream output(path, std::ios::binary);
	output.write(reinterpret_cast<const char*>(value.data()), static_cast<std::streamsize>(value.size()));
	return output.good();
}

std::string hex_digest(ByteView value)
{
	Sha256Digest digest{};
	if (!sha256(value, digest)) return {};
	std::ostringstream output;
	for (const auto byte : digest) output << std::hex << std::setfill('0') << std::setw(2) << unsigned(byte);
	return output.str();
}

std::vector<std::uint8_t> datagram(MessageType type,
	const std::vector<std::uint8_t>& payload,
	std::uint32_t sequence,
	std::uint64_t sent_time_us,
	std::uint8_t flags)
{
	TelemetryDatagramHeader header;
	header.version_minor = VersionMinorV1_1;
	header.message_type = type;
	header.flags = flags;
	header.session_id = SessionId;
	header.packet_sequence = sequence;
	header.sent_time_us = sent_time_us;
	header.message_id = sequence;
	header.fragment_count = 1U;
	header.message_size = static_cast<std::uint32_t>(payload.size());
	header.message_crc32 = crc32_iso_hdlc(bytes(payload));
	std::vector<std::uint8_t> output(HeaderSizeV1 + payload.size());
	std::size_t written = 0U;
	if (encode_datagram(header, Phase1ProducerMinorRange, bytes(payload),
			{output.data(), output.size()}, written) != ValidationError::None) {
		return {};
	}
	output.resize(written);
	return output;
}

std::vector<std::vector<std::uint8_t>> snapshot_parts(const std::vector<std::uint8_t>& source,
	bool multipart)
{
	FullSnapshotPartPayload decoded{};
	if (decode_full_snapshot_part_payload(bytes(source), decoded) != ValidationError::None) return {};
	std::vector<std::vector<std::uint8_t>> regions;
	if (!multipart) {
		regions.emplace_back(decoded.records.begin(), decoded.records.end());
	} else {
		RecordEnvelopeIterator iterator(decoded.records, decoded.record_count, RecordFlagPolicy::RequireNone);
		std::size_t split = 0U;
		for (std::uint16_t index = 0U; index < decoded.record_count / 2U; ++index) {
			RecordEnvelopeView record{};
			bool found = false;
			if (iterator.next(record, found) != ValidationError::None || !found) return {};
			split += RecordEnvelopeHeaderSize + record.payload.size;
		}
		regions.emplace_back(decoded.records.begin(), decoded.records.begin() + split);
		regions.emplace_back(decoded.records.begin() + split, decoded.records.end());
	}
	std::vector<std::vector<std::uint8_t>> result;
	for (std::size_t index = 0U; index < regions.size(); ++index) {
		FullSnapshotPartPayload part = decoded;
		part.part_index = static_cast<std::uint16_t>(index);
		part.part_count = static_cast<std::uint16_t>(regions.size());
		part.record_count = 0U;
		std::size_t offset = 0U;
		while (offset < regions[index].size()) {
			if (regions[index].size() - offset < RecordEnvelopeHeaderSize) return {};
			const auto payload_size = static_cast<std::size_t>(regions[index][offset + 4U]) |
				(static_cast<std::size_t>(regions[index][offset + 5U]) << 8U);
			const auto record_size = RecordEnvelopeHeaderSize + payload_size;
			if (record_size > regions[index].size() - offset) return {};
			++part.record_count;
			offset += record_size;
		}
		part.records = bytes(regions[index]);
		std::vector<std::uint8_t> encoded(FullSnapshotPartPayloadPrefixSize + regions[index].size());
		std::size_t written = 0U;
		if (encode_full_snapshot_part_payload(part, {encoded.data(), encoded.size()}, written) !=
			ValidationError::None) {
			return {};
		}
		result.push_back(std::move(encoded));
	}
	return result;
}

std::vector<std::vector<std::uint8_t>> manifest_parts(const fs::path& root)
{
	std::vector<std::vector<std::uint8_t>> records{
		read_binary(root / "test/telemetry/protocol/vectors/valid/records/class_manifest/class_manifest.bin"),
		read_binary(root / "test/telemetry/protocol/vectors/valid/records/weapon_manifest/weapon_manifest.bin"),
	};
	std::vector<std::uint8_t> transaction;
	for (const auto& record : records) transaction.insert(transaction.end(), record.begin(), record.end());
	Sha256Digest digest{};
	if (!sha256(bytes(transaction), digest)) return {};
	std::vector<std::vector<std::uint8_t>> result;
	for (std::size_t index = 0U; index < records.size(); ++index) {
		ManifestPartPayload part;
		part.manifest_id = 1U;
		part.part_index = static_cast<std::uint16_t>(index);
		part.part_count = static_cast<std::uint16_t>(records.size());
		part.transaction_size = static_cast<std::uint32_t>(transaction.size());
		part.transaction_sha256 = digest;
		part.producer_sample_time_us = 1'000'000U;
		part.manifest_kind = ManifestKind::FullRequired;
		part.record_count = 1U;
		part.records = bytes(records[index]);
		std::vector<std::uint8_t> encoded(ManifestPartPayloadPrefixSize + records[index].size());
		std::size_t written = 0U;
		if (encode_manifest_part_payload(part, {encoded.data(), encoded.size()}, written) !=
			ValidationError::None) {
			return {};
		}
		result.push_back(std::move(encoded));
	}
	return result;
}

struct EngineUpdateSample {
	std::string profile;
	std::vector<std::uint8_t> dto;
	std::vector<std::uint8_t> image;
	std::vector<std::vector<std::uint8_t>> wire;
	std::string authority_snapshot;
	std::string authority_class_manifest;
	std::string authority_weapon_manifest;
	bool source_stable = true;
	bool closure_stable = true;
	std::uint64_t producer_sample_time_us = 1'000'000U;
	std::uint32_t mission_generation = 1U;
	std::uint32_t manifest_id = 1U;
	std::uint32_t snapshot_id = 1U;
	std::uint64_t entity_id = 1U;
};

// Test-only sink: the producer-side DTO/image and the final wire capture are
// frozen from the same simulated EngineUpdate. It has no production linkage.
struct ProofOracleSink {
	std::string dto_sha256;
	std::string image_sha256;
	std::string wire_sha256;
	std::string authority_snapshot;
	std::string authority_class_manifest;
	std::string authority_weapon_manifest;
	bool source_stable = false;
	bool closure_stable = false;

	bool capture(const EngineUpdateSample& update)
	{
		FullSnapshotPartPayload dto{};
		StateImage canonical_image;
		const auto dto_error = decode_full_snapshot_part_payload(bytes(update.dto), dto);
		if (dto_error != ValidationError::None) {
			std::cerr << "oracle DTO decode failed: " << static_cast<unsigned>(dto_error) << '\n';
			return false;
		}
		const auto image_error =
			[&]() {
				BusinessStateValidationContext context;
				context.protocol_minor = VersionMinorV1_1;
				context.required_manifest_id = dto.required_manifest_id;
				context.class_manifest_installed = dto.required_manifest_id != 0U;
				context.weapon_manifest_installed =
					context.class_manifest_installed && update.profile == "complete-ship";
				const std::uint32_t subsystem_id = 2U;
				const BusinessClassCatalogEntry class_catalog{1U, &subsystem_id, 1U};
				if (context.class_manifest_installed) {
					context.class_catalog = &class_catalog;
					context.class_catalog_count = 1U;
				}
				BusinessStateImageValidator validator(context);
				return decode_business_snapshot_region_validated(
					dto.records, dto.record_count, validator, canonical_image);
			}();
		if (image_error != ValidationError::None) {
			std::cerr << "oracle StateImage decode failed: " << static_cast<unsigned>(image_error) << '\n';
			return false;
		}
		if (canonical_image.empty() ||
			canonical_image.encoded_snapshot_records_size() != update.image.size()) {
			std::cerr << "oracle StateImage size mismatch: "
					  << canonical_image.encoded_snapshot_records_size() << " != "
					  << update.image.size() << '\n';
			return false;
		}
		dto_sha256 = hex_digest(bytes(update.dto));
		image_sha256 = hex_digest(bytes(update.image));
		std::vector<std::uint8_t> joined;
		for (const auto& item : update.wire) joined.insert(joined.end(), item.begin(), item.end());
		wire_sha256 = hex_digest(bytes(joined));
		// The authority documents are fixed normative fixtures, not output from
		// the Python proof client. Capturing them beside the canonical C++ image
		// gives the evidence runner an independent same-EngineUpdate oracle.
		authority_snapshot = update.authority_snapshot;
		authority_class_manifest = update.authority_class_manifest;
		authority_weapon_manifest = update.authority_weapon_manifest;
		source_stable = update.source_stable;
		closure_stable = update.closure_stable;
		return !dto_sha256.empty() && !image_sha256.empty() && !wire_sha256.empty() &&
			!authority_snapshot.empty() && !authority_class_manifest.empty() &&
			!authority_weapon_manifest.empty();
	}
};

std::string json_escape(const std::string& value)
{
	std::ostringstream output;
	for (const char ch : value) {
		if (ch == '\\' || ch == '"') output << '\\';
		output << ch;
	}
	return output.str();
}

bool emit_scenario(const fs::path& root,
	const fs::path& output,
	const std::string& profile,
	bool multipart,
	std::ostringstream& scenarios,
	bool& first)
{
	const std::string vector_name =
		profile == "core-gate" ? "phase2-promotion" : "phase2-complete-ship";
	const auto snapshot = read_binary(root / "test/telemetry/protocol/vectors-v1.1" /
		vector_name / (vector_name + ".bin"));
	const auto welcome = read_binary(root /
		"test/telemetry/protocol/vectors-v1.1/welcome-accepted-minor-one/welcome-accepted-minor-one.payload.bin");
	const auto session_begin =
		read_binary(root / "test/telemetry/protocol/vectors/valid/messages/session_begin/session_begin.bin");
	const auto manifests = manifest_parts(root);
	const auto snapshots = snapshot_parts(snapshot, multipart);
	if (snapshot.empty() || welcome.empty() || session_begin.empty() || manifests.empty() || snapshots.empty()) {
		return false;
	}
	const auto name = profile + (multipart ? "-multipart" : "-minimal");
	const auto directory = output / name;
	fs::create_directories(directory);
	std::vector<std::vector<std::uint8_t>> wire;
	wire.push_back(datagram(MessageType::Welcome, welcome, 1U, 1'000'000U, MessageFlagAckRequired));
	wire.push_back(datagram(MessageType::SessionBegin, session_begin, 2U, 1'000'001U, MessageFlagAckRequired));
	std::uint32_t sequence = 3U;
	for (const auto& part : manifests) {
		wire.push_back(datagram(MessageType::Manifest, part, sequence, 1'000'000U + sequence - 1U,
			MessageFlagAckRequired));
		++sequence;
	}
	for (const auto& part : snapshots) {
		wire.push_back(datagram(MessageType::FullSnapshot, part, sequence, 1'000'000U + sequence - 1U,
			static_cast<std::uint8_t>(MessageFlagAckRequired | MessageFlagKeyframe)));
		++sequence;
	}
	for (const auto& item : wire) if (item.empty()) return false;

	EngineUpdateSample update;
	update.profile = profile;
	update.dto = snapshot;
	update.image.assign(snapshot.begin() + FullSnapshotPartPayloadPrefixSize, snapshot.end());
	update.wire = wire;
	update.authority_snapshot =
		read_text(root / "test/telemetry/protocol/expected-v1.1" / (vector_name + ".json"));
	update.authority_class_manifest =
		read_text(root / "test/telemetry/protocol/expected/records/class_manifest.json");
	update.authority_weapon_manifest =
		read_text(root / "test/telemetry/protocol/expected/records/weapon_manifest.json");
	ProofOracleSink sink;
	if (!sink.capture(update)) return false;

	std::vector<std::string> names;
	for (std::size_t index = 0U; index < wire.size(); ++index) {
		const auto filename = "datagram-" + std::to_string(index) + ".bin";
		if (!write_binary(directory / filename, wire[index])) return false;
		names.push_back(name + "/" + filename);
	}
	if (!first) scenarios << ",\n";
	first = false;
	scenarios << "    {\n"
		<< "      \"name\": \"" << json_escape(name) << "\",\n"
		<< "      \"profile\": \"" << profile << "\",\n"
		<< "      \"sample\": {\"producerSampleTimeUs\": \"1000000\", \"missionGeneration\": 1,"
		   " \"manifestId\": 1, \"snapshotId\": 1, \"entityId\": \"1\"},\n"
		<< "      \"dtoSha256\": \"" << sink.dto_sha256 << "\",\n"
		<< "      \"imageSha256\": \"" << sink.image_sha256 << "\",\n"
		<< "      \"wireSha256\": \"" << sink.wire_sha256 << "\",\n"
		<< "      \"oracleAuthority\": {\n"
		<< "        \"source\": \"fixed-canonical-cpp-engine-update\",\n"
		<< "        \"sourceStable\": " << (sink.source_stable ? "true" : "false") << ",\n"
		<< "        \"closureStable\": " << (sink.closure_stable ? "true" : "false") << ",\n"
		<< "        \"clockSample\": {\"t0\": 100000, \"t1\": 100100,"
		   " \"t2\": 100200, \"t3\": 1000000},\n"
		<< "        \"config\": {\"flightHz\": 30, \"systemsHz\": 10,"
		   " \"missionHeartbeatMs\": 500},\n"
		<< "        \"canonicalStateSha256\": \"" << sink.image_sha256 << "\",\n"
		<< "        \"snapshot\": " << sink.authority_snapshot << ",\n"
		<< "        \"manifests\": [" << sink.authority_class_manifest << ","
		<< sink.authority_weapon_manifest << "]\n"
		<< "      },\n"
		<< "      \"datagrams\": [";
	for (std::size_t index = 0U; index < names.size(); ++index) {
		if (index) scenarios << ", ";
		scenarios << "\"" << json_escape(names[index]) << "\"";
	}
	scenarios << "]\n    }";
	return true;
}

int main(int argc, char** argv)
{
	std::vector<std::string> profiles;
	fs::path output;
	for (int index = 1; index < argc; ++index) {
		const std::string argument = argv[index];
		if (argument == "--profiles") {
			while (index + 1 < argc && std::string(argv[index + 1]).rfind("--", 0U) != 0U) {
				profiles.emplace_back(argv[++index]);
			}
		} else if (argument == "--output-dir" && index + 1 < argc) {
			output = argv[++index];
		} else {
			std::cerr << "unknown or incomplete argument: " << argument << '\n';
			return 2;
		}
	}
	if (profiles.empty() || output.empty()) {
		std::cerr << "usage: telemetry_phase2_oracle_harness --profiles core-gate complete-ship --output-dir DIR\n";
		return 2;
	}
	const fs::path root = FSO_PHASE2_SOURCE_ROOT;
	fs::create_directories(output);
	std::ostringstream scenarios;
	bool first = true;
	for (const auto& profile : profiles) {
		if (profile != "core-gate" && profile != "complete-ship") {
			std::cerr << "unsupported profile: " << profile << '\n';
			return 2;
		}
		if (!emit_scenario(root, output, profile, profile == "complete-ship", scenarios, first)) {
			std::cerr << "failed to emit " << profile << " oracle fixture\n";
			return 1;
		}
	}
	std::ofstream manifest(output / "capture-manifest.json", std::ios::binary);
	manifest << "{\n  \"schema\": \"FSTL-phase2-oracle-capture-v2\",\n  \"profiles\": [";
	for (std::size_t index = 0U; index < profiles.size(); ++index) {
		if (index) manifest << ", ";
		manifest << "\"" << profiles[index] << "\"";
	}
	manifest << "],\n  \"scenarios\": [\n" << scenarios.str() << "\n  ]\n}\n";
	return manifest.good() ? 0 : 1;
}
