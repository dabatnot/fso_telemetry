using Summary = std::map<std::string, std::string>;

bool allocation_summary_passes(const Summary& summary) {
	const auto equals_value = [&](const char* key, const char* expected) {
		const auto found = summary.find(key);
		return found != summary.end() && found->second == expected;
	};
	return equals_value("noGrowth", "1") &&
		equals_value("allocationSnapshotSlot", "0") &&
		equals_value("allocationSnapshotReliable", "0") &&
		equals_value("allocationSnapshotScratch", "0") &&
		equals_value("allocationDeltaEgress", "0");
}
bool allocation_summary_oracle_self_test() {
	const Summary passing{
		{"noGrowth", "1"},
		{"allocationSnapshotSlot", "0"},
		{"allocationSnapshotReliable", "0"},
		{"allocationSnapshotScratch", "0"},
		{"allocationDeltaEgress", "0"}};
	auto growth = passing;
	growth["noGrowth"] = "0";
	auto attributed_growth = passing;
	attributed_growth["allocationSnapshotSlot"] = "1";
	auto incomplete = passing;
	incomplete.erase("allocationDeltaEgress");
	return allocation_summary_passes(passing) &&
		!allocation_summary_passes(growth) &&
		!allocation_summary_passes(attributed_growth) &&
		!allocation_summary_passes(incomplete);
}
Summary load_summary(std::filesystem::path path) {
	path.replace_extension(".summary.txt");
	std::ifstream input(path);
	Summary result;
	std::string line;
	while (std::getline(input, line)) {
		const auto equals = line.find('=');
		if (equals != std::string::npos)
			result.emplace(line.substr(0U, equals),
				line.substr(equals + 1U));
	}
	return result;
}

bool observed(const Summary& summary, const char* key) {
	const auto found = summary.find(key);
	return found != summary.end() && found->second == "1";
}

bool equals(const Summary& summary,
	const char* key,
	const char* expected) {
	const auto found = summary.find(key);
	return found != summary.end() && found->second == expected;
}

std::uint64_t summary_u64(const Summary& summary, const char* key) {
	const auto found = summary.find(key);
	if (found == summary.end()) return 0U;
	try {
		return static_cast<std::uint64_t>(std::stoull(found->second));
	} catch (...) {
		return 0U;
	}
}

bool parse_u64_arg(const char* text, std::uint64_t& value) {
	char* end = nullptr;
	value = std::strtoull(text, &end, 10);
	return text && *text && end && *end == '\0';
}

std::string escape(const std::string& value) {
	std::ostringstream out;
	for (const auto ch : value) {
		if (ch == '\\' || ch == '"') out << '\\';
		out << ch;
	}
	return out.str();
}

std::string file_sha(const std::filesystem::path& path) {
	std::ifstream input(path, std::ios::binary);
	std::vector<std::uint8_t> bytes{
		std::istreambuf_iterator<char>(input),
		std::istreambuf_iterator<char>()};
	telemetry::protocol::Sha256Digest digest{};
	if (input.bad() ||
		!telemetry::protocol::sha256(
			{bytes.data(), bytes.size()}, digest))
		return {};
	std::ostringstream output;
	for (const auto byte : digest)
		output << std::hex << std::setfill('0') << std::setw(2)
			   << static_cast<unsigned>(byte);
	return output.str();
}

std::vector<std::string> split(const std::string& line) {
	std::vector<std::string> fields;
	std::size_t begin = 0U;
	while (begin <= line.size()) {
		const auto comma = line.find(',', begin);
		fields.push_back(line.substr(begin,
			comma == std::string::npos ? comma : comma - begin));
		if (comma == std::string::npos) break;
		begin = comma + 1U;
	}
	return fields;
}

bool load_series(const std::filesystem::path& path, Series& result) {
	std::ifstream input(path);
	std::string line;
	if (!std::getline(input, line)) return false;
	constexpr auto ExpectedHeader =
		"sample_index,frame_duration_ns,telemetry_duration_ns,"
		"collect_ns,diff_ns,"
		"state_image_build_ns,state_image_fill_ns,"
		"state_image_publish_validate_ns,state_image_adopt_ns,"
		"state_image_semantic_validate_ns,delta_build_ns,"
		"serialization_ns,network_ns,allocation_events,syscall_count,"
		"queue_depth,baselines_active,is_keyframe,"
		"is_periodic_keyframe,is_forced_keyframe,trace_index,"
		"epoch_index,capture_flight,capture_systems";
	if (line != ExpectedHeader) return false;
	std::uint64_t expected_index = 0U;
	while (std::getline(input, line)) {
		const auto fields = split(line);
		if (fields.size() != 24U) return false;
		const auto index = std::strtoull(fields[0].c_str(), nullptr, 10);
		const auto trace = std::strtoull(fields[20].c_str(), nullptr, 10);
		if (index != expected_index || trace != index) return false;
		const auto tick = std::strtoull(fields[1].c_str(), nullptr, 10);
		const auto telemetry =
			std::strtoull(fields[2].c_str(), nullptr, 10);
		std::uint64_t keyframe_build = 0U;
		for (std::size_t field = 5U; field <= 9U; ++field)
			keyframe_build +=
				std::strtoull(fields[field].c_str(), nullptr, 10);
		const auto allocations =
			std::strtoull(fields[13].c_str(), nullptr, 10);
		const bool keyframe = fields[17] == "1";
		const bool periodic_keyframe = fields[18] == "1";
		const bool forced_keyframe = fields[19] == "1";
		const bool flight = fields[22] == "1";
		const bool systems = fields[23] == "1";
		if (keyframe !=
			(periodic_keyframe || forced_keyframe))
			return false;
		const auto tick_ms = static_cast<double>(tick) / 1000000.0;
		const auto telemetry_ms =
			static_cast<double>(telemetry) / 1000000.0;
		result.tick_ms.push_back(tick_ms);
		result.telemetry_ms.push_back(telemetry_ms);
		if (flight && !systems)
			result.flight_only_ms.push_back(telemetry_ms);
		if (systems) result.systems_ms.push_back(telemetry_ms);
		if (periodic_keyframe)
			result.keyframe_ms.push_back(
				static_cast<double>(keyframe_build) / 1000000.0);
		if (forced_keyframe)
			result.forced_keyframe_ms.push_back(
				static_cast<double>(keyframe_build) / 1000000.0);
		result.allocations = std::max(result.allocations, allocations);
		++result.frames;
		result.flight_ticks += flight ? 1U : 0U;
		result.systems_ticks += systems ? 1U : 0U;
		result.keyframes += periodic_keyframe ? 1U : 0U;
		result.forced_keyframes += forced_keyframe ? 1U : 0U;
		++expected_index;
	}
	return !result.tick_ms.empty();
}

double nearest_rank(std::vector<double> values, double percentile) {
	if (values.empty()) return std::numeric_limits<double>::infinity();
	std::sort(values.begin(), values.end());
	const auto rank = std::max<std::size_t>(1U,
		static_cast<std::size_t>(
			std::ceil(percentile * static_cast<double>(values.size()))));
	return values[std::min(rank, values.size()) - 1U];
}

double maximum(const std::vector<double>& values) {
	return values.empty() ? std::numeric_limits<double>::infinity()
		: *std::max_element(values.begin(), values.end());
}

double median(std::vector<double> values) {
	if (values.empty()) return std::numeric_limits<double>::infinity();
	std::sort(values.begin(), values.end());
	const auto middle = values.size() / 2U;
	return (values.size() & 1U) != 0U
		? values[middle]
		: (values[middle - 1U] + values[middle]) / 2.0;
}

bool performance_aggregation_oracle_self_test() {
	std::vector<double> active_union{1.0, 100.0, 2.0, 3.0, 4.0};
	const std::vector<double> scenario_medians{
		median({1.0, 100.0}), median({2.0, 3.0, 4.0})};
	const auto exclusive_telemetry_ns =
		[](const std::vector<std::string>& fields) {
			return std::strtoull(fields[2].c_str(), nullptr, 10);
		};
	std::vector<std::string> fields(22U, "0");
	fields[2] = "100";
	fields[3] = "40";
	fields[4] = "60";
	fields[5] = "25";
	fields[6] = "35";
	return exclusive_telemetry_ns(fields) == 100U &&
		median(active_union) == 3.0 &&
		median(scenario_medians) != median(active_union) &&
		nearest_rank({1.0, 2.0, 3.0, 100.0}, 0.99) == 100.0;
}

std::uint64_t derived_seed(std::uint64_t base, const std::string& name) {
	std::uint64_t value = base ^ 1469598103934665603ULL;
	for (const auto ch : name) {
		value ^= static_cast<unsigned char>(ch);
		value *= 1099511628211ULL;
	}
	return value;
}

void write_progress_event(const std::filesystem::path& directory,
	const std::string& stage,
	std::uint64_t completed,
	std::uint64_t total) noexcept {
	try {
		std::filesystem::create_directories(directory);
		const auto stem = std::to_string(completed) + "-" +
			std::to_string(std::chrono::duration_cast<
				std::chrono::milliseconds>(
				std::chrono::steady_clock::now().time_since_epoch())
				.count());
		const auto temporary = directory / (stem + ".tmp");
		const auto final = directory / (stem + ".json");
		std::ofstream output(temporary,
			std::ios::out | std::ios::trunc);
		output << "{\"stage\":\"" << escape(stage)
			   << "\",\"completed\":" << completed
			   << ",\"total\":" << total
			   << ",\"heartbeatMonotonicMs\":"
			   << std::chrono::duration_cast<std::chrono::milliseconds>(
					  std::chrono::steady_clock::now().time_since_epoch())
					  .count()
			   << "}\n";
		output.close();
		if (output)
			std::filesystem::rename(temporary, final);
	} catch (...) {
	}
}

class EpochCoordinator final
	: public telemetry::phase2_test::PerformanceEpochCoordinator {
  public:
	EpochCoordinator(std::uint64_t warmup_seconds,
		std::uint64_t epoch_seconds,
		std::filesystem::path progress_directory)
		: m_warmup_seconds(warmup_seconds),
		  m_epoch_seconds(epoch_seconds),
		  m_progress_directory(std::move(progress_directory)) {
		m_slice_order.reserve(EpochCount * RuntimeCount);
		write_progress_event(
			m_progress_directory, "performance", 0U,
			RuntimeCount + EpochCount * RuntimeCount);
	}

	bool begin_warmup(std::size_t slot) noexcept override {
		std::unique_lock<std::mutex> lock(m_mutex);
		m_condition.wait(lock, [&] {
			return m_cancelled || m_warmup_slot == slot;
		});
		if (m_cancelled) return false;
		m_slice_started = Clock::now();
		return true;
	}

	Clock::time_point warmup_deadline(std::size_t slot,
		std::uint64_t completed_samples,
		std::uint64_t total_samples) noexcept override {
		std::lock_guard<std::mutex> lock(m_mutex);
		if (m_cancelled || m_warmup_slot != slot || total_samples == 0U)
			return Clock::now();
		return m_slice_started +
			std::chrono::duration_cast<Clock::duration>(
				std::chrono::duration<double>(
					static_cast<double>(m_warmup_seconds) *
					static_cast<double>(completed_samples) /
					static_cast<double>(total_samples)));
	}

	void finish_warmup(std::size_t slot, bool passed) noexcept override {
		std::lock_guard<std::mutex> lock(m_mutex);
		if (!passed || slot != m_warmup_slot) {
			m_cancelled = true;
		} else {
			++m_warmup_slot;
			write_progress_event(m_progress_directory,
				"performance-warmup", m_warmup_slot,
				RuntimeCount + EpochCount * RuntimeCount);
		}
		m_condition.notify_all();
	}

	bool begin_epoch(std::size_t slot, std::size_t epoch) noexcept override {
		std::unique_lock<std::mutex> lock(m_mutex);
		m_condition.wait(lock, [&] {
			return m_cancelled ||
				(m_warmup_slot == RuntimeCount &&
				 m_epoch == epoch &&
				 expected_slot(m_epoch, m_position) == slot);
		});
		if (m_cancelled) return false;
		if (m_epoch == 0U && m_position == 0U)
			m_measured_started = Clock::now();
		m_slice_started = Clock::now();
		m_slice_order.push_back(slot);
		return true;
	}

	Clock::time_point sample_deadline(std::size_t slot,
		std::size_t epoch,
		std::uint64_t completed_samples,
		std::uint64_t samples_per_epoch) noexcept override {
		std::lock_guard<std::mutex> lock(m_mutex);
		if (m_cancelled || m_epoch != epoch ||
			expected_slot(m_epoch, m_position) != slot ||
			samples_per_epoch == 0U)
			return Clock::now();
		return m_slice_started +
			std::chrono::duration_cast<Clock::duration>(
				std::chrono::duration<double>(
					static_cast<double>(m_epoch_seconds) *
					static_cast<double>(completed_samples) /
					static_cast<double>(samples_per_epoch)));
	}

	void finish_epoch(std::size_t slot,
		std::size_t epoch,
		bool passed) noexcept override {
		std::lock_guard<std::mutex> lock(m_mutex);
		if (!passed || m_epoch != epoch ||
			expected_slot(m_epoch, m_position) != slot) {
			m_cancelled = true;
		} else if (++m_position == RuntimeCount) {
			m_position = 0U;
			if (++m_epoch == EpochCount)
				m_measured_finished = Clock::now();
		}
		if (!m_cancelled)
			write_progress_event(m_progress_directory,
				"performance-measurement",
				RuntimeCount + m_epoch * RuntimeCount + m_position,
				RuntimeCount + EpochCount * RuntimeCount);
		m_condition.notify_all();
	}

	void cancel() noexcept override {
		std::lock_guard<std::mutex> lock(m_mutex);
		m_cancelled = true;
		m_condition.notify_all();
	}

	bool passed() const noexcept {
		std::lock_guard<std::mutex> lock(m_mutex);
		return !m_cancelled && m_warmup_slot == RuntimeCount &&
			m_epoch == EpochCount &&
			m_slice_order.size() == EpochCount * RuntimeCount;
	}

	std::vector<std::size_t> slice_order() const {
		std::lock_guard<std::mutex> lock(m_mutex);
		return m_slice_order;
	}

	double measured_seconds() const noexcept {
		std::lock_guard<std::mutex> lock(m_mutex);
		if (m_measured_finished <= m_measured_started) return 0.0;
		return std::chrono::duration<double>(
			m_measured_finished - m_measured_started).count();
	}

	static std::size_t expected_slot(std::size_t epoch,
		std::size_t position) noexcept {
		return (epoch % RuntimeCount + position) % RuntimeCount;
	}

  private:
	mutable std::mutex m_mutex;
	std::condition_variable m_condition;
	std::uint64_t m_warmup_seconds = 0U;
	std::uint64_t m_epoch_seconds = 0U;
	std::size_t m_warmup_slot = 0U;
	std::size_t m_epoch = 0U;
	std::size_t m_position = 0U;
	bool m_cancelled = false;
	Clock::time_point m_slice_started{};
	Clock::time_point m_measured_started{};
	Clock::time_point m_measured_finished{};
	std::vector<std::size_t> m_slice_order;
	std::filesystem::path m_progress_directory;
};

class ProgressHeartbeat {
  public:
	ProgressHeartbeat(std::filesystem::path directory,
		std::string stage,
		std::uint64_t total)
		: m_directory(std::move(directory)),
		  m_stage(std::move(stage)),
		  m_total(total),
		  m_thread([this] { run(); }) {}
	~ProgressHeartbeat() {
		{
			std::lock_guard<std::mutex> lock(m_mutex);
			m_stopped = true;
		}
		m_condition.notify_all();
		m_thread.join();
	}
	void complete() noexcept {
		write_progress_event(m_directory, m_stage, m_total, m_total);
	}
  private:
	void run() noexcept {
		std::uint64_t heartbeat = 0U;
		write_progress_event(m_directory, m_stage, heartbeat, m_total);
		std::unique_lock<std::mutex> lock(m_mutex);
		while (!m_condition.wait_for(lock, std::chrono::minutes(10),
				[this] { return m_stopped; })) {
			lock.unlock();
			write_progress_event(
				m_directory, m_stage, ++heartbeat, m_total);
			lock.lock();
		}
	}
	std::filesystem::path m_directory;
	std::string m_stage;
	std::uint64_t m_total = 0U;
	std::mutex m_mutex;
	std::condition_variable m_condition;
	bool m_stopped = false;
	std::thread m_thread;
};

int run_native(const std::string& workload,
	const std::filesystem::path& output,
	std::uint64_t samples,
	std::uint64_t warmup,
	std::uint64_t seed,
	std::uint64_t wall_seconds,
	std::uint64_t warmup_wall_seconds,
	int coordinated_slot = -1,
	std::uint64_t epoch_samples = 0U,
	std::uint64_t epoch_count = 0U) {
	auto summary = output;
	summary.replace_extension(".summary.txt");
	std::vector<std::string> args{
		"telemetry_native_performance_runner", "--workload", workload,
		"--output", output.string(), "--summary", summary.string(),
		"--samples", std::to_string(samples),
		"--warmup", std::to_string(warmup), "--flight-hz", "30",
		"--seed", std::to_string(seed), "--diagnostic", "0",
		"--wall-seconds", std::to_string(wall_seconds),
		"--warmup-wall-seconds", std::to_string(warmup_wall_seconds)};
	if (coordinated_slot >= 0) {
		args.insert(args.end(), {
			"--coordinated-slot", std::to_string(coordinated_slot),
			"--epoch-samples", std::to_string(epoch_samples),
			"--epoch-count", std::to_string(epoch_count)});
	}
	std::vector<char*> raw;
	for (auto& arg : args) raw.push_back(arg.data());
	return telemetry_native_performance_embedded_main(
		static_cast<int>(raw.size()), raw.data());
}

}

bool verdict_log_passes(const std::filesystem::path& path) {
	if (!std::filesystem::is_regular_file(path)) return false;
	std::ifstream input(path);
	const std::string text{
		std::istreambuf_iterator<char>(input),
		std::istreambuf_iterator<char>()};
	return !text.empty() &&
		(text.find("verdict=pass") != std::string::npos ||
		 text.find("\"verdict\":\"pass\"") != std::string::npos ||
		 text.find("\"verdict\": \"pass\"") != std::string::npos);
}

bool wait_for_verdict_log(const std::filesystem::path& path) {
	for (std::size_t attempt = 0U; attempt < 100U; ++attempt) {
		if (std::filesystem::is_regular_file(path))
			return verdict_log_passes(path);
		std::this_thread::sleep_for(std::chrono::milliseconds(100));
	}
	return false;
}

int main(int argc, char** argv) {
	std::uint64_t warmup = 0U, control = 0U, measure = 0U, frames = 0U,
		flight = 0U, systems = 0U, keyframes = 0U, seed = 0U,
		soak_seconds = 0U, soak_seed = 0U;
	const char *report = nullptr, *soak_dir = nullptr,
		*soak_profile = nullptr, *qualification_plan = nullptr;
	bool smoke = false;
	bool subject_reconciliation_self_test = false;
	bool delta_scratch_capacity_self_test = false;
	bool performance_aggregation_self_test = false;
	for (int index = 1; index < argc; ++index) {
		if (!std::strcmp(argv[index], "--smoke")) {
			smoke = true;
			continue;
		}
		if (!std::strcmp(
				argv[index],
				"--subject-reconciliation-self-test")) {
			subject_reconciliation_self_test = true;
			continue;
		}
		if (!std::strcmp(
				argv[index],
				"--delta-scratch-capacity-self-test")) {
			delta_scratch_capacity_self_test = true;
			continue;
		}
		if (!std::strcmp(
				argv[index],
				"--performance-aggregation-self-test")) {
			performance_aggregation_self_test = true;
			continue;
		}
		if (index + 1 >= argc) return 2;
		const auto* key = argv[index];
		const auto* value = argv[++index];
		auto numeric = [&](std::uint64_t& target) {
			return phase2::parse_u64_arg(value, target);
		};
		if (!std::strcmp(key, "--warmup-seconds")) {
			if (!numeric(warmup)) return 2;
		} else if (!std::strcmp(key, "--control-seconds")) {
			if (!numeric(control)) return 2;
		} else if (!std::strcmp(key, "--measure-seconds")) {
			if (!numeric(measure)) return 2;
		} else if (!std::strcmp(key, "--minimum-frames")) {
			if (!numeric(frames)) return 2;
		} else if (!std::strcmp(key, "--minimum-flight-ticks")) {
			if (!numeric(flight)) return 2;
		} else if (!std::strcmp(key, "--minimum-system-ticks")) {
			if (!numeric(systems)) return 2;
		} else if (!std::strcmp(key, "--minimum-keyframes")) {
			if (!numeric(keyframes)) return 2;
		} else if (!std::strcmp(key, "--seed")) {
			if (!numeric(seed)) return 2;
		} else if (!std::strcmp(key, "--soak-seconds")) {
			if (!numeric(soak_seconds)) return 2;
		} else if (!std::strcmp(key, "--soak-seed")) {
			if (!numeric(soak_seed)) return 2;
		} else if (!std::strcmp(key, "--soak-profile")) {
			soak_profile = value;
		} else if (!std::strcmp(key, "--qualification-plan")) {
			qualification_plan = value;
		} else if (!std::strcmp(key, "--soak-report-dir")) {
			soak_dir = value;
		} else if (!std::strcmp(key, "--report")) {
			report = value;
		} else {
			return 2;
		}
	}
	if (subject_reconciliation_self_test)
		return workload_subject_reconciliation_self_test() ? 0 : 4;
	if (delta_scratch_capacity_self_test)
