    def write_raw(
        self, name: str, tick_ns: int, *,
        count: int = 600, coordinated: bool = True,
    ) -> tuple[Path, dict[str, int]]:
        path = self.root / f"{name}.csv"
        counts = {"flightTicks": 0, "systemTicks": 0, "keyframes": 0}
        with path.open("w", encoding="utf-8", newline="") as stream:
            writer = csv.DictWriter(stream, fieldnames=performance.RAW_HEADER)
            writer.writeheader()
            for index in range(count):
                flight = int(index % 2 == 0)
                systems = int(index % 6 == 0)
                keyframe = int(index % 120 == 0)
                counts["flightTicks"] += flight
                counts["systemTicks"] += systems
                counts["keyframes"] += keyframe
                row = {key: 0 for key in performance.RAW_HEADER}
                row.update({
                    "sample_index": index,
                    "trace_index": index,
                    "epoch_index": index // 60 if coordinated else 0,
                    "tick_duration_ns": tick_ns,
                    "capture_flight": flight,
                    "capture_systems": systems,
                    "is_keyframe": keyframe,
                })
                writer.writerow(row)
        return path, counts

    def test_accepts_complete_v2_fixture(self) -> None:
        performance.validate_report(self.report, self.args)

    def assert_rejected(self, mutate) -> None:
        candidate = copy.deepcopy(self.report)
        mutate(candidate)
        with self.assertRaises(ValueError):
            performance.validate_report(candidate, self.args)

    def test_rejects_wrong_rotation(self) -> None:
        self.assert_rejected(
            lambda report: report["runtimeOrder"]["slices"][5].update(
                runtime="control"
            )
        )

    def test_rejects_unproven_persistence_or_bounds(self) -> None:
        self.assert_rejected(
            lambda report: report["runtimeOrder"].update(persistent=False)
        )
        self.assert_rejected(
            lambda report: report["bounds"].update(cardinalityProven=False)
        )

    def test_rejects_raw_hash_or_aggregate_median_mismatch(self) -> None:
        self.assert_rejected(
            lambda report: report["rawSamples"][1].update(sha256="0" * 64)
        )
        self.assert_rejected(
            lambda report: report["timingsMs"].update(activeMedian=999.0)
        )

    def test_rejects_incomplete_composite_soak(self) -> None:
        self.assert_rejected(
            lambda report: report["soak"].update(clientRestarts=2)
        )
        self.assert_rejected(
            lambda report: report["soak"].update(flightTicks=299)
        )
        self.assert_rejected(
            lambda report: report["soak"].update(slotStatesExact=False)
        )

    def test_rejects_environment_log_and_bound_observation_tampering(self) -> None:
        self.assert_rejected(
            lambda report: report["host"]["temperatureLog"].update(
                sha256="0" * 64
            )
        )
        self.assert_rejected(
            lambda report: report["host"]["monitorReady"].update(
                verdict="fail"
            )
        )
        self.assert_rejected(
            lambda report: report["bounds"].update(observedShips=63)
        )

    def test_rejects_disabled_fast_path_or_clock_tampering(self) -> None:
        self.assert_rejected(
            lambda report: report["qualifications"][1].update(callbacks=99_999)
        )
        self.assert_rejected(
            lambda report: report["measurementProtocol"].update(
                clockResolutionNs=0
            )
        )

    def test_rejects_revision_command_or_build_context_tampering(self) -> None:
        self.assert_rejected(
            lambda report: report["evidenceContext"].update(revision="bad")
        )
        self.assert_rejected(
            lambda report: report["evidenceContext"].update(codeReturn=1)
        )
        self.assert_rejected(
            lambda report: report["evidenceContext"].update(buildConfig="Debug")
        )

    def test_normative_cli_contract_is_exact(self) -> None:
        args = performance.parse_args([
            "--build-dir", str(self.root),
            "--performance-report-dir", str(self.root / "performance"),
            "--soak-report-dir", str(self.root / "soak"),
        ])
        performance.validate_cli_contract(args)
        args.control_seconds = 599
        with self.assertRaises(ValueError):
            performance.validate_cli_contract(args)
