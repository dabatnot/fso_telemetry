from __future__ import annotations

import importlib.util
import json
import subprocess
import tempfile
import unittest
from pathlib import Path
from unittest import mock


SCRIPT = Path(__file__).with_name("run_phase2_reliability_evidence.py")
SPEC = importlib.util.spec_from_file_location("phase2_reliability_runner", SCRIPT)
assert SPEC is not None and SPEC.loader is not None
RUNNER = importlib.util.module_from_spec(SPEC)
SPEC.loader.exec_module(RUNNER)


class ReliabilityEvidenceContinuationTests(unittest.TestCase):
    def test_failed_cell_is_preserved_and_independent_cell_still_runs(self) -> None:
        with tempfile.TemporaryDirectory() as temporary:
            root = Path(temporary)
            harness = root / "loss-harness"
            harness.write_bytes(b"test harness")
            report_dir = root / "reports"
            calls: list[list[str]] = []

            def execute(command: list[str], **_: object) -> subprocess.CompletedProcess[str]:
                calls.append(command)
                if len(calls) == 1:
                    return subprocess.CompletedProcess(command, 4, "", "first cell failed")
                report_path = Path(command[command.index("--report") + 1])
                decision_path = Path(command[command.index("--decision-log") + 1])
                report_path.parent.mkdir(parents=True, exist_ok=True)
                decision_path.write_text("{}\n", encoding="utf-8")
                report_path.write_text(
                    json.dumps({
                        "impairment": {
                            "decisionLog": {
                                "sha256": RUNNER.sha256(decision_path),
                            },
                        },
                    }),
                    encoding="utf-8",
                )
                return subprocess.CompletedProcess(command, 0, "", "")

            with (
                mock.patch.object(RUNNER.subprocess, "run", side_effect=execute),
                mock.patch.object(RUNNER, "validate_run"),
                mock.patch.object(RUNNER, "validate_decision_log"),
            ):
                result = RUNNER.main([
                    "--cases", "core-gate:20:burst", "complete-ship:20:iid",
                    "--seed", str(RUNNER.SEED),
                    "--warmup-seconds", "0",
                    "--duration-seconds", "30",
                    "--build-dir", str(root),
                    "--report-dir", str(report_dir),
                    "--harness", str(harness),
                ])

            self.assertEqual(result, 1)
            self.assertEqual(len(calls), 2)
            manifest = json.loads(
                (report_dir / "campaign-manifest.json").read_text(encoding="utf-8")
            )
            self.assertEqual(manifest["status"], "failed")
            self.assertEqual(
                [run["status"] for run in manifest["runs"]],
                ["failed", "passed"],
            )


if __name__ == "__main__":
    unittest.main()
