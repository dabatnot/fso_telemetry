HISTORIQUE, NON NORMATIF, NE PAS EXECUTER.
[CmdletBinding()]
param(
    [string]$RawDirectory
)

Set-StrictMode -Version 2.0
$ErrorActionPreference = 'Stop'

if ([string]::IsNullOrWhiteSpace($RawDirectory)) {
    $RawDirectory = Join-Path $PSScriptRoot 'frame-benchmark'
}
$InvariantCulture = [System.Globalization.CultureInfo]::InvariantCulture
$ExpectedOrder = @(
    'control', 'phase1', 'phase1', 'control',
    'phase1', 'control', 'control', 'phase1',
    'control', 'phase1', 'phase1', 'control'
)

function Get-Median {
    param([double[]]$Values)

    if ($Values.Count -eq 0) {
        throw 'Cannot compute the median of an empty sample.'
    }

    [double[]]$Sorted = $Values | Sort-Object
    # PowerShell casts floating-point x.5 values to int with banker's rounding.
    # Use an explicit floor so odd sample counts select their true middle item.
    $Middle = [int][Math]::Floor($Sorted.Count / 2.0)
    if (($Sorted.Count % 2) -eq 1) {
        return $Sorted[$Middle]
    }

    return ($Sorted[$Middle - 1] + $Sorted[$Middle]) / 2.0
}
$OddMedianSelfCheck = Get-Median ([double[]]@(1.0, 3.0, 100.0))
$EvenMedianSelfCheck = Get-Median ([double[]]@(1.0, 3.0, 7.0, 9.0))
if ($OddMedianSelfCheck -ne 3.0 -or $EvenMedianSelfCheck -ne 5.0) {
    throw 'Internal median self-check failed.'
}

function Get-SumNs {
    param([long[]]$Values)

    [decimal]$Sum = 0
    foreach ($Value in $Values) {
        $Sum += $Value
    }
    return $Sum
}

$Files = @(Get-ChildItem -LiteralPath $RawDirectory -Filter '*.csv' | Sort-Object Name)
if ($Files.Count -ne $ExpectedOrder.Count) {
    throw "Expected 12 benchmark CSV files, found $($Files.Count)."
}

$Pooled = @{
    control = [System.Collections.Generic.List[long]]::new()
    phase1  = [System.Collections.Generic.List[long]]::new()
}
$Results = [System.Collections.Generic.List[object]]::new()

for ($FileIndex = 0; $FileIndex -lt $Files.Count; ++$FileIndex) {
    $File = $Files[$FileIndex]
    $Match = [regex]::Match($File.Name, '^run(?<run>[0-9]{2})_(?<condition>control|phase1)\.csv$')
    if (-not $Match.Success) {
        throw "Unexpected benchmark file name: $($File.Name)"
    }

    $Run = [int]$Match.Groups['run'].Value
    $Condition = $Match.Groups['condition'].Value
    if ($Run -ne ($FileIndex + 1) -or $Condition -ne $ExpectedOrder[$FileIndex]) {
        throw "Unexpected condition order at $($File.Name)."
    }

    $Timestamps = [System.Collections.Generic.List[long]]::new()
    $Durations = [System.Collections.Generic.List[long]]::new()
    foreach ($Line in Get-Content -LiteralPath $File.FullName) {
        $Fields = $Line.Split(';')
        if ($Fields.Count -ne 2) {
            throw "Malformed frame sample in $($File.Name): $Line"
        }

        $Timestamp = [long]::Parse($Fields[0], $InvariantCulture)
        $Duration = [long]::Parse($Fields[1], $InvariantCulture)
        if ($Duration -le 0) {
            throw "Non-positive frame duration in $($File.Name): $Line"
        }
        if ($Timestamps.Count -gt 0 -and $Timestamp -le $Timestamps[$Timestamps.Count - 1]) {
            throw "Non-monotonic timestamp in $($File.Name): $Line"
        }

        $Timestamps.Add($Timestamp)
        $Durations.Add($Duration)
    }

    if ($Durations.Count -le 201) {
        throw "Insufficient samples in $($File.Name): $($Durations.Count)"
    }

    # Discard the first 200 samples as per-run warm-up and the final partial
    # terminal sample. This rule is fixed for both conditions.
    [long[]]$Analyzed = $Durations.GetRange(200, $Durations.Count - 201).ToArray()
    foreach ($Duration in $Analyzed) {
        $Pooled[$Condition].Add($Duration)
    }

    [long[]]$Warmup = $Durations.GetRange(0, 200).ToArray()
    $Results.Add([pscustomobject]@{
        Run             = $Run
        Condition       = $Condition
        RawN            = $Durations.Count
        AnalyzedN       = $Analyzed.Count
        WarmupSeconds   = [double]((Get-SumNs $Warmup) / 1000000000)
        CaptureSeconds  = [double]((Get-SumNs $Durations.ToArray()) / 1000000000)
        AnalyzedSeconds = [double]((Get-SumNs $Analyzed) / 1000000000)
        MedianNs        = Get-Median $Analyzed
        Sha256          = (Get-FileHash -Algorithm SHA256 -LiteralPath $File.FullName).Hash
    })
}

foreach ($Result in $Results) {
    Write-Output ([string]::Format(
        $InvariantCulture,
        'RUN {0:D2} {1,-7} raw_n={2} analyzed_n={3} warmup_s={4:F6} capture_s={5:F6} analyzed_s={6:F6} median_ns={7:F1} sha256={8}',
        $Result.Run,
        $Result.Condition,
        $Result.RawN,
        $Result.AnalyzedN,
        $Result.WarmupSeconds,
        $Result.CaptureSeconds,
        $Result.AnalyzedSeconds,
        $Result.MedianNs,
        $Result.Sha256
    ))
}

$Summaries = @{}
foreach ($Condition in @('control', 'phase1')) {
    $ConditionResults = @($Results | Where-Object Condition -eq $Condition)
    if ($ConditionResults.Count -ne 6) {
        throw "Expected six $Condition runs, found $($ConditionResults.Count)."
    }

    [double[]]$RunMedians = @($ConditionResults | ForEach-Object { [double]$_.MedianNs })
    $Summary = [pscustomobject]@{
        Condition            = $Condition
        Runs                 = $ConditionResults.Count
        RawN                 = ($ConditionResults | Measure-Object RawN -Sum).Sum
        AnalyzedN            = ($ConditionResults | Measure-Object AnalyzedN -Sum).Sum
        CaptureSeconds       = ($ConditionResults | Measure-Object CaptureSeconds -Sum).Sum
        AnalyzedSeconds      = ($ConditionResults | Measure-Object AnalyzedSeconds -Sum).Sum
        PooledMedianNs       = Get-Median $Pooled[$Condition].ToArray()
        MedianRunMediansNs   = Get-Median $RunMedians
        MinimumRunMedianNs   = ($RunMedians | Measure-Object -Minimum).Minimum
        MaximumRunMedianNs   = ($RunMedians | Measure-Object -Maximum).Maximum
    }
    $Summaries[$Condition] = $Summary

    Write-Output ([string]::Format(
        $InvariantCulture,
        'CONDITION {0} runs={1} raw_n={2} analyzed_n={3} capture_s={4:F6} analyzed_s={5:F6} pooled_median_ns={6:F1} median_run_medians_ns={7:F1} min_run_median_ns={8:F1} max_run_median_ns={9:F1}',
        $Summary.Condition,
        $Summary.Runs,
        $Summary.RawN,
        $Summary.AnalyzedN,
        $Summary.CaptureSeconds,
        $Summary.AnalyzedSeconds,
        $Summary.PooledMedianNs,
        $Summary.MedianRunMediansNs,
        $Summary.MinimumRunMedianNs,
        $Summary.MaximumRunMedianNs
    ))
}

$PooledDeltaPercent = (($Summaries.phase1.PooledMedianNs - $Summaries.control.PooledMedianNs) / $Summaries.control.PooledMedianNs) * 100.0
$RunMedianDeltaPercent = (($Summaries.phase1.MedianRunMediansNs - $Summaries.control.MedianRunMediansNs) / $Summaries.control.MedianRunMediansNs) * 100.0
$Pass = [Math]::Abs($PooledDeltaPercent) -lt 1.0

Write-Output ([string]::Format(
    $InvariantCulture,
    'DELTA pooled_percent={0:F6} pooled_abs_percent={1:F6} run_medians_percent={2:F6} run_medians_abs_percent={3:F6} gate={4}',
    $PooledDeltaPercent,
    [Math]::Abs($PooledDeltaPercent),
    $RunMedianDeltaPercent,
    [Math]::Abs($RunMedianDeltaPercent),
    $(if ($Pass) { 'PASS' } else { 'FAIL' })
))

if (-not $Pass) {
    throw 'The absolute pooled engine-frame median delta is not below 1%.'
}
