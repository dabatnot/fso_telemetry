HISTORIQUE, NON NORMATIF, NE PAS EXECUTER.
"""Archive reproducible Phase 1 UDP impairment and lifecycle evidence.

This is deliberately only an evidence orchestrator.  The executable passed by
``--harness`` must be a test-only NativeSessionRuntime harness; this script
never claims that a CSV performance runner exercised a UDP profile.
"""

from __future__ import annotations

import argparse
import hashlib
import json
import os
import platform
import shutil
import subprocess
import sys
from datetime import datetime, timezone
from pathlib import Path
from typing import Any, Sequence


SCHEMA = "fs2open.telemetry.phase1.reliability-campaign.v1"
PROFILE_SCHEMA = "fs2open.telemetry.phase1.reliability-profile.v1"
PROFILES = (
    "independent-loss-1",
    "independent-loss-5",
    "independent-loss-20",
    "burst-loss-1",
    "burst-loss-5",
    "burst-loss-20",
)


def parse_args(argv: Sequence[str] | None = None) -> argparse.Namespace:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--harness", type=Path, required=True)
    parser.add_argument("--output-dir", type=Path, required=True)
    parser.add_argument("--duration-seconds", type=int, default=600)
    parser.add_argument("--seed", type=int, default=424242)
    parser.add_argument("--profile", action="append", choices=PROFILES)
    parser.add_argument("--restart-mission", action="store_true")
    parser.add_argument("--restart-process", action="store_true")
    parser.add_argument("--resume", action="store_true")
    return parser.parse_args(argv)


def write_json(path: Path, value: Any) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(f".{path.name}.tmp")
    temporary.write_text(json.dumps(value, indent=2, sort_keys=True) + "\n", encoding="utf-8")
    os.replace(temporary, path)


def sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def git_revision() -> str | None:
    result = subprocess.run(["git", "rev-parse", "HEAD"], text=True, capture_output=True, check=False)
    return result.stdout.strip() if result.returncode == 0 else None


def expected_impairment(profile: str) -> tuple[str, int]:
    model, _, percent = profile.rpartition("-")
    return model.replace("-loss", ""), int(percent)


def require(condition: bool, message: str) -> None:
    if not condition:
        raise ValueError(message)


def validate_report(report: Any, profile: str, seed: int, duration_seconds: int,
                    restart_mission: bool, restart_process: bool) -> None:
    require(isinstance(report, dict), "report must be a JSON object")
    require(report.get("schema") == PROFILE_SCHEMA, "wrong report schema")
    require(report.get("status") == "passed", "harness did not report passed")
    require(report.get("profile") == profile, "report profile does not match requested profile")
    require(report.get("seed") == seed, "report seed does not match command")
    require(report.get("durationSeconds") == duration_seconds, "report duration does not match command")

    loss_model, loss_percent = expected_impairment(profile)
    # v1 originally described an aspirational synthetic report shape.  The
    # shipped NativeSessionRuntime harness reports its directly observed
    # counters as ``injected`` and ``minuteObservations`` instead.  Accept
    # both shapes, but validate only fields the selected producer actually
    # measured. Native-socket transport evidence remains a separate artifact.
    injection = report.get("injection")
    if isinstance(injection, dict):
        require(injection.get("lossModel") == loss_model, "wrong loss model")
        require(injection.get("lossPercent") == loss_percent, "wrong loss percentage")
        for key in ("duplication", "reordering", "jitter", "temporaryCut", "missingFragments",
                    "slowOrSilentClient", "wouldBlock"):
            require(injection.get(key) is True, f"missing injected impairment: {key}")
        oracles = report.get("oracles")
        require(isinstance(oracles, dict), "missing oracle evidence")
        recovered = oracles.get("returnedLiveWithinSeconds")
        require(isinstance(recovered, (int, float)) and 0 <= recovered <= 10,
                "resync did not return Live within 10 seconds")
        samples = report.get("minuteSamples")
        sample_fields = ("minute", "memoryCurrentBytes", "memoryPeakBytes", "socketHandles", "queueDepth",
                         "clients", "baselines", "p99TickNs")
    else:
        configured = report.get("configured")
        injected = report.get("injected")
        convergence = report.get("convergence")
        require(isinstance(configured, dict) and isinstance(injected, dict), "missing observed impairment evidence")
        require(configured.get("lossPercent") == loss_percent, "wrong loss percentage")
        require(configured.get("burst") is (loss_model == "burst"), "wrong loss model")
        for key in ("downstreamLoss", "deltaLoss", "snapshotFragmentLoss", "outageLoss", "ackLoss",
                    "duplicateFrames", "reorderedFrames", "jitterDelayedFrames", "wouldBlock"):
            require(isinstance(injected.get(key), int) and injected[key] > 0, f"missing injected impairment: {key}")
        slow_client = report.get("slowClient")
        require(isinstance(slow_client, dict), "missing slow-client evidence")
        require(isinstance(slow_client.get("withheldReliableAcks"), int) and slow_client["withheldReliableAcks"] > 0 and
                slow_client.get("recovered") is True and slow_client.get("boundedOutcome") is True,
                "slow client did not recover through the bounded reliable path")
        require(isinstance(convergence, dict), "missing convergence evidence")
        bounded = convergence.get("boundedImpairment")
        require(isinstance(bounded, dict) and bounded.get("withinTenSecondsPostEnd") is True,
                "resync did not return Live within 10 seconds")
        recovered = bounded.get("recoveryAfterEndUs")
        require(isinstance(recovered, int) and 0 < recovered <= 10_000_000,
                "invalid post-impairment recovery")
        samples = report.get("minuteObservations")
        sample_fields = ("minute", "memoryBytes", "sockets", "queueDepth", "clients", "baselines", "p99TickNs")

    require(isinstance(samples, list) and len(samples) >= max(1, (duration_seconds + 59) // 60),
            "missing per-minute soak observations")
    for sample in samples:
        require(isinstance(sample, dict), "invalid minute sample")
        for key in sample_fields:
            require(isinstance(sample.get(key), int) and sample[key] >= 0, f"invalid minute sample field: {key}")

    lifecycle = report.get("lifecycle")
    require(isinstance(lifecycle, dict), "missing lifecycle evidence")
    if restart_mission:
        require((lifecycle.get("missionGenerationIncreased") is True and lifecycle.get("missionSessionRenewed") is True) or
                (lifecycle.get("missionConverged") is True and lifecycle.get("missionGenerationAfter") > lifecycle.get("missionGenerationBefore") and lifecycle.get("missionSessionId") != lifecycle.get("initialSessionId")),
                "mission restart evidence is incomplete")
    if restart_process:
        require((lifecycle.get("processSessionChanged") is True and lifecycle.get("noLiveModuleAllocationsAfterShutdown") is True) or
                (lifecycle.get("processConverged") is True and lifecycle.get("processSessionId") != lifecycle.get("initialSessionId") and lifecycle.get("processShutdownReleased") is True),
                "process restart/leak evidence is incomplete")


def run_profile(args: argparse.Namespace, profile: str, index: int, campaign: dict[str, Any]) -> bool:
    profile_dir = args.output_dir / profile
    report_path = profile_dir / "harness-report.json"
    stdout_path = profile_dir / "stdout.log"
    stderr_path = profile_dir / "stderr.log"
    seed = args.seed + index
    executable = [str(args.harness)] if args.harness.suffix.lower() != ".py" else [sys.executable, str(args.harness)]
    command = [*executable, "--profile", profile, "--seed", str(seed), "--duration-seconds",
               str(args.duration_seconds), "--report", str(report_path)]
    if args.restart_mission:
        command.append("--restart-mission")
    if args.restart_process:
        command.append("--restart-process")
    record: dict[str, Any] = {"profile": profile, "seed": seed, "command": command, "status": "running",
                               "exitCode": None, "report": str(report_path), "error": None}
    campaign["profiles"].append(record)
    write_json(args.output_dir / "campaign-manifest.json", campaign)
    if report_path.exists() and not args.resume:
        record.update(status="failed", error="report already exists; use --resume only for a valid report")
        write_json(args.output_dir / "campaign-manifest.json", campaign)
        return False
    profile_dir.mkdir(parents=True, exist_ok=True)
    try:
        with stdout_path.open("w", encoding="utf-8", newline="\n") as stdout, stderr_path.open("w", encoding="utf-8", newline="\n") as stderr:
            completed = subprocess.run(command, stdout=stdout, stderr=stderr, check=False)
        record["exitCode"] = completed.returncode
        require(completed.returncode == 0, f"harness exit code {completed.returncode}")
        report = json.loads(report_path.read_text(encoding="utf-8"))
        validate_report(report, profile, seed, args.duration_seconds, args.restart_mission, args.restart_process)
        record.update(status="passed", reportSha256=sha256(report_path))
        write_json(profile_dir / "validated-report.json", report)
    except (OSError, ValueError, json.JSONDecodeError) as error:
        record.update(status="failed", error=str(error))
    write_json(args.output_dir / "campaign-manifest.json", campaign)
    return record["status"] == "passed"


def main(argv: Sequence[str] | None = None) -> int:
    args = parse_args(argv)
    if args.duration_seconds <= 0 or args.seed < 0:
        print("duration-seconds must be positive and seed must be non-negative", file=sys.stderr)
        return 2
    if not args.harness.is_file():
        print(f"missing WP11 test-only harness: {args.harness}", file=sys.stderr)
        return 2
    selected = tuple(args.profile) if args.profile else PROFILES
    if len(set(selected)) != len(selected):
        print("a profile may be selected only once", file=sys.stderr)
        return 2
    args.output_dir.mkdir(parents=True, exist_ok=True)
    manifest_path = args.output_dir / "campaign-manifest.json"
    if manifest_path.exists() and not args.resume:
        print(f"campaign manifest already exists: {manifest_path}", file=sys.stderr)
        return 2
    campaign: dict[str, Any] = {
        "schema": SCHEMA, "status": "running", "revision": git_revision(), "startedAtUtc": datetime.now(timezone.utc).isoformat(),
        "harness": str(args.harness.resolve()), "harnessSha256": sha256(args.harness), "durationSeconds": args.duration_seconds,
        "seed": args.seed, "restartMission": args.restart_mission, "restartProcess": args.restart_process,
        "platform": platform.platform(), "python": sys.version, "profiles": [],
    }
    write_json(manifest_path, campaign)
    results = [run_profile(args, profile, index, campaign) for index, profile in enumerate(selected)]
    succeeded = all(results)
    campaign["status"] = "passed" if succeeded else "failed"
    campaign["completedAtUtc"] = datetime.now(timezone.utc).isoformat()
    write_json(manifest_path, campaign)
    return 0 if succeeded else 1


if __name__ == "__main__":
    raise SystemExit(main())
